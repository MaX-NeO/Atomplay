    64→interface QuestionFormState {
    65→  questionText: string
    66→  optionA: string
    67→  optionB: string
    68→  optionC: string
    69→  optionD: string
    70→  correctOption: OptionKey
    71→  timeLimit: number
    72→}
    73→
    74→function toFormState(q: QuestionDTO): QuestionFormState {
    75→  return {
    76→    questionText: q.questionText,
    77→    optionA: q.optionA,
    78→    optionB: q.optionB,
    79→    optionC: q.optionC,
    80→    optionD: q.optionD,
    81→    correctOption: q.correctOption,
    82→    timeLimit: q.timeLimit,
    83→  }
    84→}
    85→
    86→function isFormValid(f: QuestionFormState | null): boolean {
    87→  if (!f) return false
    88→  return (
    89→    f.questionText.trim().length > 0 &&
    90→    f.optionA.trim().length > 0 &&
    91→    f.optionB.trim().length > 0 &&
    92→    f.optionC.trim().length > 0 &&
    93→    f.optionD.trim().length > 0
    94→  )
    95→}

    97→export function ActivityEditorScreen() {
    98→  const navigate = useAppStore((s) => s.navigate)
    99→  const params = useAppStore((s) => s.params)
   100→  const activityId = params.activityId
   101→
   102→  const [activity, setActivity] = useState<ActivityDTO | null>(null)
   103→  const [loading, setLoading] = useState(true)
   104→  const [selectedId, setSelectedId] = useState<string | null>(null)
   105→  const [form, setForm] = useState<QuestionFormState | null>(null)
   106→  const [dirty, setDirty] = useState(false)
   107→  const [saving, setSaving] = useState(false)
   108→  const [publishOpen, setPublishOpen] = useState(false)
   109→  const [publishing, setPublishing] = useState(false)
   110→  const [publishResult, setPublishResult] = useState<ActivityDTO | null>(null)
   111→  const [deletingQuestion, setDeletingQuestion] = useState<QuestionDTO | null>(null)
   112→  const [titleDraft, setTitleDraft] = useState('')
   113→  const [savingTitle, setSavingTitle] = useState(false)
   114→  const titleInputRef = useRef<HTMLInputElement>(null)
   115→
   116→  const fetchActivity = useCallback(async () => {
   117→    if (!activityId) return
   118→    setLoading(true)
   119→    try {
   120→      const res = await api.get<{ activity: ActivityDTO }>(`/api/activities/${activityId}`)
   121→      setActivity(res.activity)
   122→      setTitleDraft(res.activity.title)
   123→      setSelectedId((prev) => {
   124→        const exists = res.activity.questions?.some((q) => q.id === prev)
   125→        if (exists) return prev
   126→        return res.activity.questions?.[0]?.id ?? null
   127→      })
   128→    } catch (err) {
   129→      const msg = err instanceof ApiError ? err.message : 'Failed to load activity'
   130→      toast.error(msg)
   131→      if (err instanceof ApiError && err.status === 401) {
   132→        navigate('admin-login')
   133→      } else if (err instanceof ApiError && err.status === 404) {
   134→        navigate('admin-dashboard')
   135→      }
   136→    } finally {
   137→      setLoading(false)
   138→    }
   139→  }, [activityId, navigate])
   140→
   141→  useEffect(() => {
   142→    void fetchActivity()
   143→  }, [fetchActivity])
   144→
   145→  // Sync form when selectedId changes.
   146→  useEffect(() => {
   147→    if (!activity || !selectedId) {
   148→      setForm(null)
   149→      setDirty(false)
   150→      return
   151→    }
   152→    const q = activity.questions?.find((x) => x.id === selectedId)
   153→    if (q) {
   154→      setForm(toFormState(q))
   155→      setDirty(false)
   156→    }
   157→  }, [selectedId, activity])
   158→
   159→  const selectedQuestion = useMemo(() => {
   160→    if (!activity || !selectedId) return null
   161→    return activity.questions?.find((q) => q.id === selectedId) ?? null
   162→  }, [activity, selectedId])
   163→
   164→  function updateField<K extends keyof QuestionFormState>(field: K, value: QuestionFormState[K]) {
   165→    if (!form) return
   166→    setForm({ ...form, [field]: value })
   167→    setDirty(true)
   168→  }
   169→
   170→  async function handleSaveQuestion() {
   171→    if (!form || !selectedQuestion) return
   172→    if (!isFormValid(form)) {
   173→      toast.error('Question text and all 4 options are required')
   174→      return
   175→    }
   176→    setSaving(true)
   177→    try {
   178→      const res = await api.patch<{ question: QuestionDTO }>(
   179→        `/api/questions/${selectedQuestion.id}`,
   180→        {
   181→          questionText: form.questionText,
   182→          optionA: form.optionA,
   183→          optionB: form.optionB,
   184→          optionC: form.optionC,
   185→          optionD: form.optionD,
   186→          correctOption: form.correctOption,
   187→          timeLimit: form.timeLimit,
   188→        },
   189→      )
   190→      setActivity((prev) =>
   191→        prev
   192→          ? {
   193→              ...prev,
   194→              questions: prev.questions?.map((q) =>
   195→                q.id === res.question.id ? res.question : q,
   196→              ),
   197→            }
   198→          : prev,
   199→      )
   200→      setDirty(false)
   201→      toast.success('Question saved')
   202→    } catch (err) {
   203→      const msg = err instanceof ApiError ? err.message : 'Failed to save question'
   204→      toast.error(msg)
   205→    } finally {
   206→      setSaving(false)
   207→    }
   208→  }
   209→
   210→  async function handleAddQuestion() {
   211→    if (!activity) return
   212→    try {
   213→      const res = await api.post<{ question: QuestionDTO }>(
   214→        `/api/activities/${activity.id}/questions`,
   215→        {
   216→          questionText: 'New question',
   217→          optionA: 'Option A',
   218→          optionB: 'Option B',
   219→          optionC: 'Option C',
   220→          optionD: 'Option D',
   221→          correctOption: 'A',
   222→          timeLimit: 30,
   223→        },
   224→      )
   225→      setActivity((prev) =>
   226→        prev
   227→          ? { ...prev, questions: [...(prev.questions ?? []), res.question] }
   228→          : prev,
   229→      )
   230→      setSelectedId(res.question.id)
   231→      toast.success('Question added')
   232→    } catch (err) {
   233→      const msg = err instanceof ApiError ? err.message : 'Failed to add question'
   234→      toast.error(msg)
   235→    }
   236→  }
   237→
   238→  async function handleDeleteQuestion() {
   239→    if (!deletingQuestion || !activity) return
   240→    const targetId = deletingQuestion.id
   241→    try {
   242→      await api.delete(`/api/questions/${targetId}`)
   243→      const remaining = (activity.questions ?? []).filter((q) => q.id !== targetId)
   244→      setActivity({ ...activity, questions: remaining })
   245→      if (selectedId === targetId) {
   246→        setSelectedId(remaining[0]?.id ?? null)
   247→      }
   248→      toast.success('Question deleted')
   249→    } catch (err) {
   250→      const msg = err instanceof ApiError ? err.message : 'Failed to delete question'
   251→      toast.error(msg)
   252→    } finally {
   253→      setDeletingQuestion(null)
   254→    }
   255→  }
   256→
   257→  async function handleSaveTitle() {
   258→    if (!activity) return
   259→    if (titleDraft.trim() === activity.title) return
   260→    if (!titleDraft.trim()) {
   261→      setTitleDraft(activity.title)
   262→      return
   263→    }
   264→    setSavingTitle(true)
   265→    try {
   266→      const res = await api.patch<{ activity: ActivityDTO }>(`/api/activities/${activity.id}`, {
   267→        title: titleDraft.trim(),
   268→      })
   269→      setActivity((prev) => (prev ? { ...prev, title: res.activity.title } : prev))
   270→      toast.success('Title updated')
   271→    } catch (err) {
   272→      const msg = err instanceof ApiError ? err.message : 'Failed to update title'
   273→      toast.error(msg)
   274→      setTitleDraft(activity.title)
   275→    } finally {
   276→      setSavingTitle(false)
   277→    }
   278→  }
   279→
   280→  function handleTitleKeyDown(e: KeyboardEvent<HTMLInputElement>) {
   281→    if (e.key === 'Enter') {
   282→      e.preventDefault()
   283→      e.currentTarget.blur()
   284→    }
   285→  }
   286→
   287→  async function handlePublish() {
   288→    if (!activity) return
   289→    setPublishing(true)
   290→    try {
   291→      const res = await api.post<{ activity: ActivityDTO }>(
   292→        `/api/activities/${activity.id}/publish`,
   293→      )
   294→      setActivity(res.activity)
   295→      setPublishResult(res.activity)
   296→      toast.success('Activity published!')
   297→    } catch (err) {
   298→      const msg = err instanceof ApiError ? err.message : 'Failed to publish'
   299→      toast.error(msg)
   300→    } finally {
   301→      setPublishing(false)
   302→    }
   303→  }
   304→
   305→  function handleTitleSubmit(e: FormEvent) {
   306→    e.preventDefault()
   307→    titleInputRef.current?.blur()
   308→  }
   309→
   310→  async function copyAccessCode(code: string) {
   311→    try {
   312→      await navigator.clipboard.writeText(code)
   313→      toast.success('Access code copied')
   314→    } catch {
   315→      toast.error('Could not copy code')
   316→    }
   317→  }
   318→
   319→  // ---------- Loading ----------
   320→  if (loading) {
   321→    return (
   322→      <Shell>
   323→        <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
   324→          <Skeleton className="mb-6 h-12 w-72" />
   325→          <div className="grid gap-6 lg:grid-cols-[300px_1fr]">
   326→            <div className="space-y-3">
   327→              {Array.from({ length: 3 }).map((_, i) => (
   328→                <Skeleton key={i} className="h-16 w-full" />
   329→              ))}
   330→            </div>
   331→            <Skeleton className="h-96 w-full" />
   332→          </div>
   333→        </div>
   334→      </Shell>
   335→    )
   336→  }
   337→
   338→  if (!activity) {
   339→    return (
   340→      <Shell>
   341→        <div className="mx-auto max-w-7xl px-4 py-12 text-center sm:px-6 lg:px-8">
   342→          <p className="text-muted-foreground">Activity not found.</p>
   343→          <Button variant="ghost" onClick={() => navigate('admin-dashboard')} className="mt-4">
   344→            <ArrowLeft className="h-4 w-4" /> Back to dashboard
   345→          </Button>
   346→        </div>
   347→      </Shell>
   348→    )
   349→  }
   350→
   351→  const questionCount = activity.questions?.length ?? 0
   352→  const formInvalid = !isFormValid(form)
   353→
   354→  // The SELECTED question is locked only when the activity is LIVE and this
   355→  // question is the one currently being presented (activity.currentQuestionId).
   356→  // All other questions — and all non-LIVE activities — are fully editable.
   357→  const isLiveQuestion =
   358→    activity.status === 'LIVE' &&
   359→    activity.currentQuestionId != null &&
   360→    selectedQuestion?.id === activity.currentQuestionId
   361→
   362→  return (
   363→    <Shell>
   364→      {/* Top bar */}
   365→      <header className="sticky top-0 z-30 border-b border-border/60 bg-background/80 backdrop-blur">
   366→        <div className="mx-auto flex max-w-7xl items-center gap-3 px-4 py-3 sm:px-6 lg:px-8">
   367→          <Button
   368→            variant="ghost"
   369→            size="icon"
   370→            onClick={() => navigate('admin-dashboard')}
   371→            aria-label="Back to dashboard"
   372→          >
   373→            <ArrowLeft className="h-4 w-4" />
   374→          </Button>
   375→          <form className="relative min-w-0 flex-1" onSubmit={handleTitleSubmit}>
   376→            <Input
   377→              ref={titleInputRef}
   378→              value={titleDraft}
   379→              onChange={(e) => setTitleDraft(e.target.value)}
   380→              onBlur={handleSaveTitle}
   381→              onKeyDown={handleTitleKeyDown}
   382→              disabled={savingTitle}
   383→              className="h-10 border-transparent bg-transparent px-2 text-base font-semibold hover:border-input focus-visible:border-input sm:text-lg"
   384→              aria-label="Activity title"
   385→            />
   386→            {savingTitle && (
   387→              <Loader2 className="absolute right-2 top-1/2 h-3.5 w-3.5 -translate-y-1/2 animate-spin text-muted-foreground" />
   388→            )}
   389→          </form>
   390→          <Badge variant="outline" className={STATUS_BADGE_CLASS[activity.status]}>
   391→            {activity.status}
   392→          </Badge>
   393→          {/* Recommended max participants — DISPLAY ONLY. The join API
   394→              (`src/app/api/join/route.ts`) actually accepts up to 99, but the
   395→              admin-facing recommendation is 80 so hosts plan their room size
   396→              accordingly. See MAX_PARTICIPANTS_DISPLAY in
   397→              `src/lib/participant-icons.tsx`. */}
   398→          <Badge
   399→            variant="outline"
   400→            className="hidden items-center gap-1.5 border-primary/30 bg-primary/5 text-primary sm:inline-flex"
   401→            title={`Recommended room size — the join API accepts up to 99 participants`}
   402→          >
   403→            <Users className="h-3.5 w-3.5" />
   404→            Max {MAX_PARTICIPANTS_DISPLAY} participants
   405→          </Badge>
   406→          <div className="hidden items-center gap-2 sm:flex">
   407→            {activity.status === 'DRAFT' ? (
   408→              <Button
   409→                onClick={() => setPublishOpen(true)}
   410→                disabled={questionCount === 0 || publishing}
   411→                size="sm"
   412→              >
   413→                <Sparkles className="h-4 w-4" /> Publish
   414→              </Button>
   415→            ) : (
   416→              <div className="flex items-center gap-1 rounded-md border border-border bg-muted/40 px-2 py-1">
   417→                <span className="font-mono text-sm font-bold tracking-widest">
   418→                  {activity.accessCode ?? '------'}
   419→                </span>
   420→                <Button
   421→                  variant="ghost"
   422→                  size="icon"
   423→                  className="h-7 w-7"
   424→                  onClick={() => activity.accessCode && copyAccessCode(activity.accessCode)}
   425→                  aria-label="Copy access code"
   426→                >
   427→                  <Copy className="h-3.5 w-3.5" />
   428→                </Button>
   429→              </div>
   430→            )}
   431→            <Button
   432→              variant="secondary"
   433→              size="sm"
   434→              disabled={activity.status === 'DRAFT'}
   435→              onClick={() => navigate('admin-present', { activityId: activity.id })}
   436→            >
   437→              <Presentation className="h-4 w-4" /> Present
   438→            </Button>
   439→          </div>
   440→        </div>
   441→        {/* Mobile actions */}
   442→        <div className="flex items-center gap-2 border-t border-border/40 px-4 py-2 sm:hidden">
   443→          {activity.status === 'DRAFT' ? (
   444→            <Button
   445→              onClick={() => setPublishOpen(true)}
   446→              disabled={questionCount === 0 || publishing}
   447→              size="sm"
   448→              className="flex-1"
   449→            >
   450→              <Sparkles className="h-4 w-4" /> Publish
   451→            </Button>
   452→          ) : (
   453→            <Button
   454→              variant="outline"
   455→              size="sm"
   456→              className="flex-1"
   457→              onClick={() => activity.accessCode && copyAccessCode(activity.accessCode)}
   458→            >
   459→              <span className="font-mono font-bold tracking-widest">{activity.accessCode}</span>
   460→              <Copy className="h-3.5 w-3.5" />
   461→            </Button>
   462→          )}
   463→          <Button
   464→            variant="secondary"
   465→            size="sm"
   466→            disabled={activity.status === 'DRAFT'}
   467→            onClick={() => navigate('admin-present', { activityId: activity.id })}
   468→          >
   469→            <Presentation className="h-4 w-4" />
   470→          </Button>
   471→        </div>
   472→      </header>
   473→
   474→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-6 sm:px-6 lg:px-8">
   475→        <div className="grid gap-6 lg:grid-cols-[300px_1fr]">
   476→          {/* Question list */}
   477→          <aside className="lg:sticky lg:top-28 lg:self-start">
   478→            <div className="mb-2 flex items-center justify-between">
   479→              <h2 className="text-sm font-semibold text-muted-foreground">
   480→                Questions ({questionCount})
   481→              </h2>
   482→            </div>
   483→            <ScrollArea className="h-[calc(100vh-14rem)] pr-3">
   484→              <div className="space-y-2">
   485→                {(activity.questions ?? []).map((q, i) => {
   486→                  const isSelected = q.id === selectedId
   487→                  return (
   488→                    <button
   489→                      key={q.id}
   490→                      onClick={() => setSelectedId(q.id)}
   491→                      className={`group relative flex w-full items-start gap-3 rounded-xl border-2 px-3 py-2.5 text-left transition-all ${
   492→                        isSelected
   493→                          ? 'border-primary bg-primary/5'
   494→                          : 'border-border bg-card hover:border-primary/40 hover:bg-accent/40'
   495→                      }`}
   496→                    >
   497→                      <span
   498→                        className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-xs font-bold ${
   499→                          isSelected
   500→                            ? 'bg-primary text-primary-foreground'
   501→                            : 'bg-muted text-muted-foreground'
   502→                        }`}
   503→                      >
   504→                        {i + 1}
   505→                      </span>
   506→                      <div className="min-w-0 flex-1">
   507→                        <p className="truncate text-sm font-medium">
   508→                          {q.questionText || (
   509→                            <span className="italic text-muted-foreground">Untitled question</span>
   510→                          )}
   511→                        </p>
   512→                        <p className="mt-0.5 text-xs text-muted-foreground">
   513→                          {q.timeLimit}s · correct: {q.correctOption}
   514→                        </p>
   515→                      </div>
   516→                      {!(activity.status === 'LIVE' && activity.currentQuestionId === q.id) && (
   517→                        <span
   518→                          role="button"
   519→                          tabIndex={0}
   520→                          aria-label={`Delete question ${i + 1}`}
   521→                          onClick={(e) => {
   522→                            e.stopPropagation()
   523→                            setDeletingQuestion(q)
   524→                          }}
   525→                          onKeyDown={(e) => {
   526→                            if (e.key === 'Enter' || e.key === ' ') {
   527→                              e.preventDefault()
   528→                              e.stopPropagation()
   529→                              setDeletingQuestion(q)
   530→                            }
   531→                          }}
   532→                          className="absolute right-1.5 top-1.5 hidden h-7 w-7 items-center justify-center rounded-md text-muted-foreground hover:bg-destructive/10 hover:text-destructive group-hover:flex"
   533→                        >
   534→                          <Trash2 className="h-3.5 w-3.5" />
   535→                        </span>
   536→                      )}
   537→                    </button>
   538→                  )
   539→                })}
   540→                {questionCount === 0 && (
   541→                  <p className="px-3 py-6 text-center text-xs text-muted-foreground">
   542→                    No questions yet. Add your first one!
   543→                  </p>
   544→                )}
   545→              </div>
   546→              <Button
   547→                onClick={handleAddQuestion}
   548→                variant="outline"
   549→                className="mt-3 w-full border-dashed"
   550→              >
   551→                <Plus className="h-4 w-4" /> Add question
   552→              </Button>
   553→            </ScrollArea>
   554→          </aside>
   555→
   556→          {/* Editor */}
   557→          <section>
   558→            <AnimatePresence mode="wait">
   559→              {selectedQuestion && form ? (
   560→                <motion.div
   561→                  key={selectedQuestion.id}
   562→                  initial={{ opacity: 0, y: 10 }}
   563→                  animate={{ opacity: 1, y: 0 }}
   564→                  exit={{ opacity: 0, y: -10 }}
   565→                  transition={{ duration: 0.2 }}
   566→                >
   567→                  <Card>
   568→                    <CardHeader className="flex flex-row items-start justify-between gap-2 space-y-0">
   569→                      <div>
   570→                        <CardTitle className="flex items-center gap-2">
   571→                          <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-primary/15 text-sm font-bold text-primary">
   572→                            {selectedQuestion.questionOrder}
   573→                          </span>
   574→                          Edit question
   575→                        </CardTitle>
   576→                        <CardDescription>
   577→                          {isLiveQuestion
   578→                            ? 'This question is currently live — end it on the presentation screen before editing.'
   579→                            : 'Keep it clear and concise. Tap a letter badge to mark the correct answer.'}
   580→                        </CardDescription>
   581→                      </div>
   582→                      {isLiveQuestion ? (
   583→                        <Badge
   584→                          variant="outline"
   585→                          className="border-rose-500/40 text-rose-600 dark:text-rose-400"
   586→                        >
   587→                          <Radio className="mr-1 h-3 w-3" /> Live
   588→                        </Badge>
   589→                      ) : dirty ? (
   590→                        <Badge
   591→                          variant="outline"
   592→                          className="border-amber-500/40 text-amber-600 dark:text-amber-400"
   593→                        >
   594→                          Unsaved changes
   595→                        </Badge>
   596→                      ) : null}
   597→                    </CardHeader>
   598→                    <CardContent className="space-y-6">
   599→                      <div className="space-y-2">
   600→                        <Label htmlFor="questionText">Question</Label>
   601→                        <Textarea
   602→                          id="questionText"
   603→                          value={form.questionText}
   604→                          onChange={(e) => updateField('questionText', e.target.value)}
   605→                          placeholder="Type your question…"
   606→                          rows={3}
   607→                          disabled={isLiveQuestion}
   608→                        />
   609→                      </div>
   610→
   611→                      <div className="space-y-3">
   612→                        <Label>Options</Label>
   613→                        <div className="space-y-2">
   614→                          {OPTION_KEYS.map((key) => {
   615→                            const field = `option${key}` as
   616→                              | 'optionA'
   617→                              | 'optionB'
   618→                              | 'optionC'
   619→                              | 'optionD'
   620→                            const isCorrect = form.correctOption === key
   621→                            return (
   622→                              <div
   623→                                key={key}
   624→                                className={`flex items-center gap-2 rounded-xl border-2 px-2 py-1.5 transition-colors ${
   625→                                  isCorrect
   626→                                    ? 'border-emerald-500/60 bg-emerald-500/5'
   627→                                    : 'border-border bg-card'
   628→                                }`}
   629→                              >
   630→                                <button
   631→                                  type="button"
   632→                                  onClick={() => updateField('correctOption', key)}
   633→                                  disabled={isLiveQuestion}
   634→                                  aria-label={`Mark option ${key} as correct`}
   635→                                  aria-pressed={isCorrect}
   636→                                  className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-lg text-sm font-bold transition-colors ${
   637→                                    isCorrect
   638→                                      ? 'bg-emerald-500 text-white'
   639→                                      : 'bg-muted text-muted-foreground hover:bg-primary/15 hover:text-primary'
   640→                                  }`}
   641→                                >
   642→                                  {isCorrect ? <Check className="h-4 w-4" /> : key}
   643→                                </button>
   644→                                <Input
   645→                                  value={form[field]}
   646→                                  onChange={(e) => updateField(field, e.target.value)}
   647→                                  placeholder={`Option ${key}`}
   648→                                  disabled={isLiveQuestion}
   649→                                  className="h-10 border-0 bg-transparent shadow-none focus-visible:ring-0"
   650→                                />
   651→                              </div>
   652→                            )
   653→                          })}
   654→                        </div>
   655→                        <p className="text-xs text-muted-foreground">
   656→                          Correct answer:{' '}
   657→                          <span className="font-semibold text-emerald-600 dark:text-emerald-400">
   658→                            {form.correctOption}
   659→                          </span>
   660→                        </p>
   661→                      </div>
   662→
   663→                      <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
   664→                        <div className="space-y-2">
   665→                          <Label htmlFor="timeLimit">Time limit</Label>
   666→                          <Select
   667→                            value={String(form.timeLimit)}
   668→                            onValueChange={(v) => updateField('timeLimit', Number(v))}
   669→                            disabled={isLiveQuestion}
   670→                          >
   671→                            <SelectTrigger id="timeLimit" className="w-40">
   672→                              <SelectValue />
   673→                            </SelectTrigger>
   674→                            <SelectContent>
   675→                              {TIME_LIMITS.map((t) => (
   676→                                <SelectItem key={t} value={String(t)}>
   677→                                  {t} seconds
   678→                                </SelectItem>
   679→                              ))}
   680→                            </SelectContent>
   681→                          </Select>
   682→                        </div>
   683→
   684→                        <div className="flex items-center gap-2">
   685→                          {!isLiveQuestion && (
   686→                            <Button
   687→                              variant="ghost"
   688→                              onClick={() => setDeletingQuestion(selectedQuestion)}
   689→                              className="text-destructive hover:bg-destructive/10 hover:text-destructive"
   690→                            >
   691→                              <Trash2 className="h-4 w-4" /> Delete
   692→                            </Button>
   693→                          )}
   694→                          <Button
   695→                            onClick={handleSaveQuestion}
   696→                            disabled={isLiveQuestion || formInvalid || saving || !dirty}
   697→                          >
   698→                            {saving ? (
   699→                              <Loader2 className="h-4 w-4 animate-spin" />
   700→                            ) : (
   701→                              <Save className="h-4 w-4" />
   702→                            )}
   703→                            Save
   704→                          </Button>
   705→                        </div>
   706→                      </div>
   707→                    </CardContent>
   708→                  </Card>
   709→                </motion.div>
   710→              ) : (
   711→                <motion.div
   712→                  key="empty"
   713→                  initial={{ opacity: 0 }}
   714→                  animate={{ opacity: 1 }}
   715→                  exit={{ opacity: 0 }}
   716→                  className="flex h-64 flex-col items-center justify-center rounded-2xl border-2 border-dashed border-border/60 bg-card/40 text-center"
   717→                >
   718→                  <Sparkles className="h-10 w-10 text-muted-foreground/60" />
   719→                  <p className="mt-3 text-sm text-muted-foreground">
   720→                    Select a question on the left to edit, or add a new one.
   721→                  </p>
   722→                  <Button onClick={handleAddQuestion} variant="outline" className="mt-4">
   723→                    <Plus className="h-4 w-4" /> Add first question
   724→                  </Button>
   725→                </motion.div>
   726→              )}
   727→            </AnimatePresence>
   728→          </section>
   729→        </div>
   730→      </main>
   731→
   732→      {/* Publish dialog */}
   733→      <Dialog
   734→        open={publishOpen}
   735→        onOpenChange={(o) => {
   736→          setPublishOpen(o)
   737→          if (!o) setPublishResult(null)
   738→        }}
   739→      >
   740→        <DialogContent>
   741→          {publishResult ? (
   742→            <>
   743→              <DialogHeader>
   744→                <DialogTitle>Activity published!</DialogTitle>
   745→                <DialogDescription>
   746→                  Share this access code with your participants. They can join from the participant entrance.
   747→                </DialogDescription>
   748→              </DialogHeader>
   749→              <div className="rounded-2xl border-2 border-dashed border-primary/30 bg-primary/5 p-6 text-center">
   750→                <p className="text-xs font-semibold uppercase tracking-widest text-muted-foreground">
   751→                  Access code
   752→                </p>
   753→                <p className="mt-2 font-mono text-5xl font-bold tracking-[0.3em] text-primary">
   754→                  {publishResult.accessCode}
   755→                </p>
   756→                <Button
   757→                  variant="outline"
   758→                  size="sm"
   759→                  className="mt-3"
   760→                  onClick={() =>
   761→                    publishResult.accessCode && copyAccessCode(publishResult.accessCode)
   762→                  }
   763→                >
   764→                  <Copy className="h-3.5 w-3.5" /> Copy code
   765→                </Button>
   766→              </div>
   767→              <DialogFooter className="gap-2">
   768→                <Button variant="ghost" onClick={() => setPublishOpen(false)}>
   769→                  Stay in editor
   770→                </Button>
   771→                <Button
   772→                  onClick={() => navigate('admin-present', { activityId: publishResult.id })}
   773→                >
   774→                  <Presentation className="h-4 w-4" /> Present now
   775→                </Button>
   776→              </DialogFooter>
   777→            </>
   778→          ) : (
   779→            <>
   780→              <DialogHeader>
   781→                <DialogTitle>Publish activity?</DialogTitle>
   782→                <DialogDescription>
   783→                  This will generate a 6-digit access code so participants can join. You can still
   784→                  edit the title, questions, and options after publishing.
   785→                </DialogDescription>
   786→              </DialogHeader>
   787→              <div className="rounded-xl bg-muted/40 p-3 text-sm">
   788→                <div className="flex items-center justify-between">
   789→                  <span className="text-muted-foreground">Title</span>
   790→                  <span className="truncate pl-2 text-right font-medium">{activity.title}</span>
   791→                </div>
   792→                <Separator className="my-2" />
   793→                <div className="flex items-center justify-between">
   794→                  <span className="text-muted-foreground">Questions</span>
   795→                  <span className="font-medium">{questionCount}</span>
   796→                </div>
   797→                <Separator className="my-2" />
   798→                <div className="flex items-center justify-between">
   799→                  <span className="text-muted-foreground">Max participants</span>
   800→                  <span className="font-medium">{MAX_PARTICIPANTS_DISPLAY}</span>
   801→                </div>
   802→              </div>
   803→              <DialogFooter className="gap-2">
   804→                <Button variant="ghost" onClick={() => setPublishOpen(false)} disabled={publishing}>
   805→                  Cancel
   806→                </Button>
   807→                <Button onClick={handlePublish} disabled={publishing}>
   808→                  {publishing ? (
   809→                    <Loader2 className="h-4 w-4 animate-spin" />
   810→                  ) : (
   811→                    <Sparkles className="h-4 w-4" />
   812→                  )}
   813→                  Publish now
   814→                </Button>
   815→              </DialogFooter>
   816→            </>
   817→          )}
   818→        </DialogContent>
   819→      </Dialog>
   820→
   821→      {/* Delete question confirm */}
   822→      <AlertDialog
   823→        open={!!deletingQuestion}
   824→        onOpenChange={(o) => !o && setDeletingQuestion(null)}
   825→      >
   826→        <AlertDialogContent>
   827→          <AlertDialogHeader>
   828→            <AlertDialogTitle>
   829→              Delete question {deletingQuestion?.questionOrder}?
   830→            </AlertDialogTitle>
   831→            <AlertDialogDescription>
   832→              This cannot be undone. Remaining questions will be renumbered automatically.
   833→            </AlertDialogDescription>
   834→          </AlertDialogHeader>
   835→          <AlertDialogFooter>
   836→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   837→            <AlertDialogAction
   838→              onClick={handleDeleteQuestion}
   839→              className="bg-destructive text-white hover:bg-destructive/90"
   840→            >
   841→              Delete
   842→            </AlertDialogAction>
   843→          </AlertDialogFooter>
   844→        </AlertDialogContent>
   845→      </AlertDialog>
   846→    </Shell>
   847→  )
   848→}
   849→
   850→function Shell({ children }: { children: React.ReactNode }) {
