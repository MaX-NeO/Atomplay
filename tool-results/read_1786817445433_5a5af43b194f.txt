    92→function StatusBadge({ status }: { status: ActivityStatus }) {
    93→  if (status === 'LIVE') {
    94→    return (
    95→      <Badge className="gap-1.5 rounded-full border-amber-500/30 bg-amber-500/15 text-amber-600 dark:text-amber-400">
    96→        <span className="relative flex h-2 w-2">
    97→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-amber-500/80" />
    98→          <span className="relative inline-flex h-2 w-2 rounded-full bg-amber-500" />
    99→        </span>
   100→        LIVE
   101→      </Badge>
   102→    )
   103→  }
   104→  if (status === 'PUBLISHED') {
   105→    return (
   106→      <Badge className="gap-1.5 rounded-full border-emerald-500/30 bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   107→        <CircleDot className="h-3 w-3" />
   108→        Published
   109→      </Badge>
   110→    )
   111→  }
   112→  if (status === 'COMPLETED') {
   113→    return (
   114→      <Badge
   115→        variant="secondary"
   116→        className="gap-1.5 rounded-full bg-muted text-muted-foreground"
   117→      >
   118→        Completed
   119→      </Badge>
   120→    )
   121→  }
   122→  return (
   123→    <Badge
   124→      variant="secondary"
   125→      className="gap-1.5 rounded-full bg-muted text-muted-foreground"
   126→    >
   127→      <FileText className="h-3 w-3" />
   128→      Draft
   129→    </Badge>
   130→  )
   131→}

   133→function initials(name: string) {
   134→  return name
   135→    .split(' ')
   136→    .map((n) => n[0])
   137→    .slice(0, 2)
   138→    .join('')
   139→    .toUpperCase()
   140→}

   142→function greeting() {
   143→  const h = new Date().getHours()
   144→  if (h < 12) return 'morning'
   145→  if (h < 18) return 'afternoon'
   146→  return 'evening'
   147→}

   149→export function AdminDashboardScreen() {
   150→  const admin = useAppStore((s) => s.admin)
   151→  const navigate = useAppStore((s) => s.navigate)
   152→  const setAdmin = useAppStore((s) => s.setAdmin)
   153→  const toggleTheme = useAppStore((s) => s.toggleTheme)
   154→  const theme = useAppStore((s) => s.theme)
   155→
   156→  const [booting, setBooting] = useState(true)
   157→  const [activities, setActivities] = useState<ActivityWithCount[]>([])
   158→  const [loading, setLoading] = useState(true)
   159→  const [search, setSearch] = useState('')
   160→  const [filter, setFilter] = useState<FilterKey>('all')
   161→
   162→  // Create dialog
   163→  const [createOpen, setCreateOpen] = useState(false)
   164→  const [newTitle, setNewTitle] = useState('')
   165→  const [newDesc, setNewDesc] = useState('')
   166→  const [creating, setCreating] = useState(false)
   167→
   168→  // Delete confirm
   169→  const [deleteTarget, setDeleteTarget] = useState<ActivityWithCount | null>(null)
   170→  const [deleting, setDeleting] = useState(false)
   171→
   172→  // View access code
   173→  const [codeActivity, setCodeActivity] = useState<ActivityWithCount | null>(null)
   174→
   175→  useEffect(() => {
   176→    // Give the page-level bootstrap a brief moment to hydrate admin state.
   177→    const t = setTimeout(() => setBooting(false), 400)
   178→    return () => clearTimeout(t)
   179→  }, [])
   180→
   181→  useEffect(() => {
   182→    if (booting) return
   183→    if (!admin) {
   184→      navigate('admin-login')
   185→      return
   186→    }
   187→    void fetchActivities()
   188→  }, [admin, booting, navigate])
   189→
   190→  async function fetchActivities() {
   191→    setLoading(true)
   192→    try {
   193→      const res = await api.get<{ activities: ActivityWithCount[] }>(
   194→        '/api/activities',
   195→      )
   196→      setActivities(res.activities)
   197→    } catch {
   198→      toast.error('Failed to load activities')
   199→    } finally {
   200→      setLoading(false)
   201→    }
   202→  }
   203→
   204→  const filtered = useMemo(() => {
   205→    return activities.filter((a) => {
   206→      if (filter !== 'all' && a.status !== filter) return false
   207→      if (search && !a.title.toLowerCase().includes(search.toLowerCase()))
   208→        return false
   209→      return true
   210→    })
   211→  }, [activities, filter, search])
   212→
   213→  const stats = useMemo(() => {
   214→    return {
   215→      total: activities.length,
   216→      liveOrPublished: activities.filter(
   217→        (a) => a.status === 'PUBLISHED' || a.status === 'LIVE',
   218→      ).length,
   219→      participants: activities.reduce(
   220→        (sum, a) => sum + (a._count?.participants ?? 0),
   221→        0,
   222→      ),
   223→    }
   224→  }, [activities])
   225→
   226→  async function handleCreate() {
   227→    if (!newTitle.trim()) return
   228→    setCreating(true)
   229→    try {
   230→      const res = await api.post<{ activity: ActivityDTO }>('/api/activities', {
   231→        title: newTitle.trim(),
   232→        description: newDesc.trim(),
   233→      })
   234→      toast.success('Activity created')
   235→      setCreateOpen(false)
   236→      setNewTitle('')
   237→      setNewDesc('')
   238→      navigate('admin-editor', { activityId: res.activity.id })
   239→    } catch (err) {
   240→      const msg = err instanceof ApiError ? err.message : 'Failed to create activity'
   241→      toast.error(msg)
   242→    } finally {
   243→      setCreating(false)
   244→    }
   245→  }
   246→
   247→  async function handleDelete() {
   248→    if (!deleteTarget) return
   249→    setDeleting(true)
   250→    try {
   251→      await api.delete(`/api/activities/${deleteTarget.id}`)
   252→      toast.success(`“${deleteTarget.title}” deleted`)
   253→      setActivities((prev) => prev.filter((a) => a.id !== deleteTarget.id))
   254→    } catch (err) {
   255→      const msg = err instanceof ApiError ? err.message : 'Failed to delete activity'
   256→      toast.error(msg)
   257→    } finally {
   258→      setDeleting(false)
   259→      setDeleteTarget(null)
   260→    }
   261→  }
   262→
   263→  async function handleSignOut() {
   264→    try {
   265→      await api.post('/api/auth/logout')
   266→    } catch {
   267→      /* ignore */
   268→    }
   269→    setAdmin(null)
   270→    toast.success('Signed out')
   271→    navigate('landing')
   272→  }
   273→
   274→  if (booting) {
   275→    return (
   276→      <div className="flex min-h-screen flex-col bg-background">
   277→        <main className="flex flex-1 items-center justify-center">
   278→          <Loader2 className="h-6 w-6 animate-spin text-primary" />
   279→        </main>
   280→        <AppFooter />
   281→      </div>
   282→    )
   283→  }
   284→
   285→  if (!admin) return null
   286→
   287→  return (
   288→    <div className="flex min-h-screen flex-col bg-background">
   289→      {/* Top nav */}
   290→      <header className="sticky top-0 z-30 border-b border-border/40 bg-background/80 backdrop-blur-md">
   291→        <div className="mx-auto flex h-16 max-w-7xl items-center justify-between px-4 sm:px-6 lg:px-8">
   292→          <button
   293→            type="button"
   294→            onClick={() => navigate('landing')}
   295→            className="flex items-center gap-2.5"
   296→          >
   297→            <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-primary/15 text-primary">
   298→              <Sparkles className="h-5 w-5" />
   299→            </div>
   300→            <span className="text-lg font-semibold tracking-tight">Atom Play</span>
   301→          </button>
   302→
   303→          <div className="flex items-center gap-2">
   304→            <Button
   305→              variant="ghost"
   306→              size="icon"
   307→              onClick={toggleTheme}
   308→              aria-label="Toggle theme"
   309→              className="h-10 w-10 rounded-xl"
   310→            >
   311→              {theme === 'dark' ? (
   312→                <Sun className="h-5 w-5" />
   313→              ) : (
   314→                <Moon className="h-5 w-5" />
   315→              )}
   316→            </Button>
   317→            <DropdownMenu>
   318→              <DropdownMenuTrigger asChild>
   319→                <Button
   320→                  variant="ghost"
   321→                  className="h-10 gap-2 rounded-xl px-2 pr-3"
   322→                  aria-label="Account menu"
   323→                >
   324→                  <Avatar className="h-7 w-7">
   325→                    <AvatarFallback className="bg-primary/15 text-xs font-semibold text-primary">
   326→                      {initials(admin.name)}
   327→                    </AvatarFallback>
   328→                  </Avatar>
   329→                  <span className="hidden text-sm font-medium sm:inline">
   330→                    {admin.name}
   331→                  </span>
   332→                  <ChevronDown className="h-4 w-4 text-muted-foreground" />
   333→                </Button>
   334→              </DropdownMenuTrigger>
   335→              <DropdownMenuContent align="end" className="w-56 rounded-xl">
   336→                <DropdownMenuLabel className="flex flex-col gap-0.5">
   337→                  <span>{admin.name}</span>
   338→                  <span className="text-xs font-normal text-muted-foreground">
   339→                    {admin.email}
   340→                  </span>
   341→                </DropdownMenuLabel>
   342→                <DropdownMenuSeparator />
   343→                <DropdownMenuItem onClick={() => navigate('admin-dashboard')}>
   344→                  <LayoutDashboard className="h-4 w-4" />
   345→                  Dashboard
   346→                </DropdownMenuItem>
   347→                {admin.role === 'SUPER_ADMIN' && (
   348→                  <DropdownMenuItem onClick={() => navigate('admin-admins')}>
   349→                    <UserCog className="h-4 w-4" />
   350→                    Admin Management
   351→                  </DropdownMenuItem>
   352→                )}
   353→                <DropdownMenuSeparator />
   354→                <DropdownMenuItem
   355→                  onClick={handleSignOut}
   356→                  variant="destructive"
   357→                >
   358→                  <LogOut className="h-4 w-4" />
   359→                  Sign out
   360→                </DropdownMenuItem>
   361→              </DropdownMenuContent>
   362→            </DropdownMenu>
   363→          </div>
   364→        </div>
   365→      </header>
   366→
   367→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-8 sm:px-6 lg:px-8">
   368→        {/* Greeting + create */}
   369→        <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
   370→          <div>
   371→            <h1 className="text-2xl font-bold tracking-tight sm:text-3xl">
   372→              Good {greeting()},{' '}
   373→              <span className="gradient-text">{admin.name.split(' ')[0]}</span>
   374→            </h1>
   375→            <p className="mt-1 text-sm text-muted-foreground">
   376→              Manage your quizzes, go live, and review audience responses.
   377→            </p>
   378→          </div>
   379→          <Button
   380→            onClick={() => setCreateOpen(true)}
   381→            className="h-11 rounded-xl px-5"
   382→            size="lg"
   383→          >
   384→            <Plus className="h-4 w-4" />
   385→            Create Activity
   386→          </Button>
   387→        </div>
   388→
   389→        {/* Stats */}
   390→        <section
   391→          aria-label="Stats"
   392→          className="mt-8 grid gap-4 sm:grid-cols-3"
   393→        >
   394→          <StatCard
   395→            label="Total activities"
   396→            value={stats.total}
   397→            icon={FileText}
   398→            tint="text-chart-1"
   399→            bg="bg-chart-1/10"
   400→            loading={loading}
   401→          />
   402→          <StatCard
   403→            label="Published & Live"
   404→            value={stats.liveOrPublished}
   405→            icon={Radio}
   406→            tint="text-chart-2"
   407→            bg="bg-chart-2/10"
   408→            loading={loading}
   409→          />
   410→          <StatCard
   411→            label="Total participants"
   412→            value={stats.participants}
   413→            icon={Users}
   414→            tint="text-chart-3"
   415→            bg="bg-chart-3/10"
   416→            loading={loading}
   417→          />
   418→        </section>
   419→
   420→        {/* Filters */}
   421→        <section
   422→          aria-label="Activities"
   423→          className="mt-8 space-y-4"
   424→        >
   425→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   426→            <Tabs
   427→              value={filter}
   428→              onValueChange={(v) => setFilter(v as FilterKey)}
   429→            >
   430→              <TabsList className="h-10 rounded-lg overflow-x-auto">
   431→                {FILTERS.map((f) => (
   432→                  <TabsTrigger key={f.key} value={f.key} className="px-3">
   433→                    {f.label}
   434→                  </TabsTrigger>
   435→                ))}
   436→              </TabsList>
   437→            </Tabs>
   438→            <div className="relative w-full sm:w-72">
   439→              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   440→              <Input
   441→                placeholder="Search by title…"
   442→                value={search}
   443→                onChange={(e) => setSearch(e.target.value)}
   444→                className="h-10 rounded-lg pl-9"
   445→                aria-label="Search activities"
   446→              />
   447→            </div>
   448→          </div>
   449→
   450→          {/* List */}
   451→          {loading ? (
   452→            <div className="grid gap-4">
   453→              {Array.from({ length: 3 }).map((_, i) => (
   454→                <Card key={i} className="rounded-2xl">
   455→                  <CardContent className="flex flex-col gap-3 py-5">
   456→                    <Skeleton className="h-5 w-1/3" />
   457→                    <Skeleton className="h-4 w-2/3" />
   458→                    <div className="flex gap-2 pt-2">
   459→                      <Skeleton className="h-8 w-20 rounded-lg" />
   460→                      <Skeleton className="h-8 w-20 rounded-lg" />
   461→                    </div>
   462→                  </CardContent>
   463→                </Card>
   464→              ))}
   465→            </div>
   466→          ) : filtered.length === 0 ? (
   467→            <EmptyState onCreate={() => setCreateOpen(true)} hasAny={activities.length > 0} />
   468→          ) : (
   469→            <div className="grid max-h-[60vh] gap-4 overflow-y-auto scroll-thin pr-1">
   470→              <AnimatePresence initial={false}>
   471→                {filtered.map((activity) => (
   472→                  <motion.div
   473→                    key={activity.id}
   474→                    layout
   475→                    initial={{ opacity: 0, y: 8 }}
   476→                    animate={{ opacity: 1, y: 0 }}
   477→                    exit={{ opacity: 0, y: -8 }}
   478→                    transition={{ duration: 0.2 }}
   479→                  >
   480→                    <ActivityCard
   481→                      activity={activity}
   482→                      onEdit={() =>
   483→                        navigate('admin-editor', { activityId: activity.id })
   484→                      }
   485→                      onPresent={() =>
   486→                        navigate('admin-present', { activityId: activity.id })
   487→                      }
   488→                      onResults={() =>
   489→                        navigate('admin-results', { activityId: activity.id })
   490→                      }
   491→                      onViewCode={() => setCodeActivity(activity)}
   492→                      onDelete={() => setDeleteTarget(activity)}
   493→                    />
   494→                  </motion.div>
   495→                ))}
   496→              </AnimatePresence>
   497→            </div>
   498→          )}
   499→        </section>
   500→      </main>
   501→
   502→      <AppFooter />
   503→
   504→      {/* Create dialog */}
   505→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   506→        <DialogContent className="rounded-2xl">
   507→          <DialogHeader>
   508→            <DialogTitle>Create a new activity</DialogTitle>
   509→            <DialogDescription>
   510→              Give your quiz a name and an optional description. You can add
   511→              questions next.
   512→            </DialogDescription>
   513→          </DialogHeader>
   514→          <div className="space-y-4">
   515→            <div className="space-y-2">
   516→              <Label htmlFor="act-title">Title</Label>
   517→              <Input
   518→                id="act-title"
   519→                placeholder="e.g. Week 3 trivia — World Capitals"
   520→                value={newTitle}
   521→                onChange={(e) => setNewTitle(e.target.value)}
   522→                className="h-11 rounded-lg"
   523→                autoFocus
   524→              />
   525→            </div>
   526→            <div className="space-y-2">
   527→              <Label htmlFor="act-desc">Description (optional)</Label>
   528→              <Textarea
   529→                id="act-desc"
   530→                placeholder="A short summary of what this quiz is about."
   531→                value={newDesc}
   532→                onChange={(e) => setNewDesc(e.target.value)}
   533→                className="rounded-lg"
   534→              />
   535→            </div>
   536→          </div>
   537→          <DialogFooter>
   538→            <Button
   539→              variant="outline"
   540→              onClick={() => setCreateOpen(false)}
   541→              disabled={creating}
   542→              className="rounded-lg"
   543→            >
   544→              Cancel
   545→            </Button>
   546→            <Button
   547→              onClick={handleCreate}
   548→              disabled={!newTitle.trim() || creating}
   549→              className="rounded-lg"
   550→            >
   551→              {creating ? (
   552→                <>
   553→                  <Loader2 className="h-4 w-4 animate-spin" />
   554→                  Creating…
   555→                </>
   556→              ) : (
   557→                <>
   558→                  <Plus className="h-4 w-4" />
   559→                  Create & edit
   560→                </>
   561→              )}
   562→            </Button>
   563→          </DialogFooter>
   564→        </DialogContent>
   565→      </Dialog>
   566→
   567→      {/* View access code dialog */}
   568→      <Dialog
   569→        open={!!codeActivity}
   570→        onOpenChange={(open) => !open && setCodeActivity(null)}
   571→      >
   572→        <DialogContent className="rounded-2xl">
   573→          <DialogHeader>
   574→            <DialogTitle>Access code</DialogTitle>
   575→            <DialogDescription>
   576→              Share this 6-digit code with your participants. They can join
   577→              from the landing page.
   578→            </DialogDescription>
   579→          </DialogHeader>
   580→          {codeActivity && (
   581→            <div className="flex flex-col items-center gap-4 py-2">
   582→              <div className="flex items-center gap-3 rounded-2xl border border-dashed border-border/70 bg-muted/30 px-6 py-5">
   583→                <QrCode className="h-10 w-10 text-primary" />
   584→                <span className="font-mono text-5xl font-bold tracking-[0.3em] text-foreground">
   585→                  {codeActivity.accessCode ?? '—'}
   586→                </span>
   587→              </div>
   588→              <p className="text-center text-sm text-muted-foreground">
   589→                {codeActivity.title}
   590→              </p>
   591→            </div>
   592→          )}
   593→          <DialogFooter>
   594→            <Button
   595→              onClick={() => {
   596→                if (codeActivity?.accessCode) {
   597→                  navigator.clipboard
   598→                    ?.writeText(codeActivity.accessCode)
   599→                    .then(() => toast.success('Code copied'))
   600→                    .catch(() => toast.error('Copy failed'))
   601→                }
   602→              }}
   603→              className="rounded-lg"
   604→              disabled={!codeActivity?.accessCode}
   605→            >
   606→              Copy code
   607→            </Button>
   608→          </DialogFooter>
   609→        </DialogContent>
   610→      </Dialog>
   611→
   612→      {/* Delete confirm */}
   613→      <AlertDialog
   614→        open={!!deleteTarget}
   615→        onOpenChange={(open) => !open && setDeleteTarget(null)}
   616→      >
   617→        <AlertDialogContent className="rounded-2xl">
   618→          <AlertDialogHeader>
   619→            <AlertDialogTitle>Delete activity?</AlertDialogTitle>
   620→            <AlertDialogDescription>
   621→              {deleteTarget && (
