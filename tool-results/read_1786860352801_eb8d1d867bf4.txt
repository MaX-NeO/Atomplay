    69→// 70-color palette generated by hue rotation around the full color wheel.
    70→// Each entry has a vivid border + matching soft glow + readable text color.
    71→const PARTICIPANT_COLORS: { border: string; glow: string; text: string; soft: string }[] =
    72→  Array.from({ length: 70 }, (_, i) => {
    73→    const hue = Math.round((i * 360) / 70)
    74→    return {
    75→      border: `hsl(${hue} 85% 65%)`,
    76→      glow: `hsl(${hue} 85% 55% / 0.45)`,
    77→      text: `hsl(${hue} 90% 82%)`,
    78→      soft: `hsl(${hue} 85% 55% / 0.16)`,
    79→    }
    80→  })

    82→// Deterministic color assignment — mix the name hash with the index so visually
    83→// nearby bubbles don't always clash, and the same person always gets the same
    84→// color across reloads.
    85→function colorForParticipant(name: string, idx: number) {
    86→  let hash = 0
    87→  for (let i = 0; i < name.length; i++) hash = (hash * 31 + name.charCodeAt(i)) | 0
    88→  return PARTICIPANT_COLORS[Math.abs(hash + idx * 7) % PARTICIPANT_COLORS.length]
    89→}

    91→// Phyllotaxis "sunflower" layout — the first participant is pinned to the
    92→// center as a large anchor bubble, and each subsequent joiner spawns around
    93→// them at the golden angle (137.5°). The cluster stays balanced as it grows
    94→// and every outside bubble gets its own size + independent drift animation
    95→// so the whole field shimmers gently instead of rigidly rotating.
    96→const GOLDEN_ANGLE = Math.PI * (3 - Math.sqrt(5)) // ~2.39996 rad = 137.5°
    97→
    98→const CENTER_BUBBLE_DIAMETER = 132 // first user — large anchor
    99→const OUTSIDE_BUBBLE_MIN = 44 // smallest outside circle
   100→const OUTSIDE_BUBBLE_MAX = 72 // largest outside circle
   101→const DRIFT_AMPLITUDE = 6 // ±px movement for outside bubbles
   102→
   103→function phyllotaxisPosition(i: number): { x: number; y: number } {
   104→  if (i === 0) return { x: 0, y: 0 } // first user — dead center
   105→  const r = 78 + 24 * Math.sqrt(i)
   106→  const theta = i * GOLDEN_ANGLE
   107→  return { x: Math.cos(theta) * r, y: Math.sin(theta) * r }
   108→}
   109→
   110→// Deterministic pseudo-random in [0, 1) from a string seed — keeps each
   111→// bubble's size + drift stable across reloads and re-renders.
   112→function seededRand(seed: string): number {
   113→  let h = 0
   114→  for (let i = 0; i < seed.length; i++) h = (h * 33 + seed.charCodeAt(i)) | 0
   115→  return (Math.abs(h) % 10000) / 10000
   116→}
   117→
   118→function bubbleDiameter(name: string, idx: number): number {
   119→  const r = seededRand(`${name}-${idx}-size`)
   120→  return OUTSIDE_BUBBLE_MIN + r * (OUTSIDE_BUBBLE_MAX - OUTSIDE_BUBBLE_MIN)
   121→}
   122→
   123→// Each outside bubble drifts on its own tiny loop so the cluster feels alive
   124→// without any big rotation. Returns x/y keyframes + loop timing.
   125→function bubbleDrift(name: string, idx: number) {
   126→  const r1 = seededRand(`${name}-${idx}-x1`)
   127→  const r2 = seededRand(`${name}-${idx}-x2`)
   128→  const r3 = seededRand(`${name}-${idx}-y1`)
   129→  const r4 = seededRand(`${name}-${idx}-y2`)
   130→  const r5 = seededRand(`${name}-${idx}-dur`)
   131→  const r6 = seededRand(`${name}-${idx}-delay`)
   132→  return {
   133→    x: [0, (r1 - 0.5) * 2 * DRIFT_AMPLITUDE, 0, (r2 - 0.5) * 2 * DRIFT_AMPLITUDE, 0],
   134→    y: [0, (r3 - 0.5) * 2 * DRIFT_AMPLITUDE, 0, (r4 - 0.5) * 2 * DRIFT_AMPLITUDE, 0],
   135→    duration: 5 + r5 * 4, // 5..9s
   136→    delay: r6 * 2, // 0..2s
   137→  }
   138→}

   140→// Adaptive scale — keeps the phyllotaxis cluster fitting inside the viewport as
   141→// the participant count grows. Computed from the outermost bubble's distance
   142→// + max bubble size, clamped to [0.4, 1] so it never gets unreadable or huge.
   143→function computeClusterScale(count: number, vw: number, vh: number): number {
   144→  const outer = 78 + 24 * Math.sqrt(Math.max(0, count - 1)) + OUTSIDE_BUBBLE_MAX / 2
   145→  const diameter = outer * 2
   146→  // Leave room for header (~180px) + start button (~90px) + side padding.
   147→  const avail = Math.min(vw - 64, vh - 280)
   148→  if (avail <= 0) return 0.5
   149→  const raw = avail / diameter
   150→  return Math.max(0.4, Math.min(1, raw))
   151→}

   160→function buildLiveFromDTO(q: QuestionDTO, total: number): LiveQuestion {
   161→  return {
   162→    questionId: q.id,
   163→    questionOrder: q.questionOrder,
   164→    totalQuestions: total,
   165→    questionText: q.questionText,
   166→    options: [
   167→      { key: 'A', label: q.optionA },
   168→      { key: 'B', label: q.optionB },
   169→      { key: 'C', label: q.optionC },
   170→      { key: 'D', label: q.optionD },
   171→    ],
   172→    endsAt: '', // populated by caller using activity.questionEndsAt
   173→  }
   174→}

   176→function labelsFromLive(q: LiveQuestion | null): Partial<Record<OptionKey, string>> {
   177→  if (!q) return {}
   178→  const out: Partial<Record<OptionKey, string>> = {}
   179→  for (const o of q.options) out[o.key] = o.label
   180→  return out
   181→}
