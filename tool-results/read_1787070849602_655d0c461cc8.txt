    74→    74→function toFormState(q: QuestionDTO): QuestionFormState {

    83→    83→  }
    84→    84→}

    86→    86→function isFormValid(f: QuestionFormState | null): boolean {
    87→    87→  if (!f) return false
    88→    88→  return (
    89→    89→    f.questionText.trim().length > 0 &&
    90→    90→    f.optionA.trim().length > 0 &&
    91→    91→    f.optionB.trim().length > 0 &&
    92→    92→    f.optionC.trim().length > 0 &&
    93→    93→    f.optionD.trim().length > 0
    94→    94→  )
    95→    95→}

   124→   124→        const exists = res.activity.questions?.some((q) => q.id === prev)

   132→   132→        navigate('admin-login')

   134→   134→        navigate('admin-dashboard')
   135→   135→      }
   136→   136→    } finally {
   137→   137→      setLoading(false)
   138→   138→    }
   139→   139→  }, [activityId, navigate])
   140→   140→
   141→   141→  useEffect(() => {
   142→   142→    void fetchActivity()

   144→   144→
   145→   145→  // Sync form when selectedId changes.

   148→   148→      setForm(null)
   149→   149→      setDirty(false)

   158→   158→
   159→   159→  const selectedQuestion = useMemo(() => {

   164→   164→  function updateField<K extends keyof QuestionFormState>(field: K, value: QuestionFormState[K]) {

   169→   169→

   173→   173→      toast.error('Question text and all 4 options are required')

   179→   179→        `/api/questions/${selectedQuestion.id}`,
   180→   180→        {
   181→   181→          questionText: form.questionText,
   182→   182→          optionA: form.optionA,
   183→   183→          optionB: form.optionB,
   184→   184→          optionC: form.optionC,
   185→   185→          optionD: form.optionD,
   186→   186→          correctOption: form.correctOption,
   187→   187→          timeLimit: form.timeLimit,
   188→   188→        },

   190→   190→      setActivity((prev) =>
   191→   191→        prev
   192→   192→          ? {
   193→   193→              ...prev,
   194→   194→              questions: prev.questions?.map((q) =>
   195→   195→                q.id === res.question.id ? res.question : q,
   196→   196→              ),
   197→   197→            }
   198→   198→          : prev,
   199→   199→      )
   200→   200→      setDirty(false)
   201→   201→      toast.success('Question saved')
   202→   202→    } catch (err) {
   203→   203→      const msg = err instanceof ApiError ? err.message : 'Failed to save question'
   204→   204→      toast.error(msg)
   205→   205→    } finally {
   206→   206→      setSaving(false)
   207→   207→    }
   208→   208→  }

   238→   238→  async function handleDeleteQuestion() {

   242→   242→      await api.delete(`/api/questions/${targetId}`)

   253→   253→      setDeletingQuestion(null)
   254→   254→    }

   257→   257→  async function handleSaveTitle() {

   276→   276→      setSavingTitle(false)
