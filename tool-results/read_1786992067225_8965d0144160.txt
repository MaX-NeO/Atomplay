    63→interface QuestionFormState {
    64→  questionText: string
    65→  optionA: string
    66→  optionB: string
    67→  optionC: string
    68→  optionD: string
    69→  correctOption: OptionKey
    70→  timeLimit: number
    71→}
    72→
    73→function toFormState(q: QuestionDTO): QuestionFormState {
    74→  return {
    75→    questionText: q.questionText,
    76→    optionA: q.optionA,
    77→    optionB: q.optionB,
    78→    optionC: q.optionC,
    79→    optionD: q.optionD,
    80→    correctOption: q.correctOption,
    81→    timeLimit: q.timeLimit,
    82→  }
    83→}
    84→
    85→function isFormValid(f: QuestionFormState | null): boolean {
    86→  if (!f) return false
    87→  return (
    88→    f.questionText.trim().length > 0 &&
    89→    f.optionA.trim().length > 0 &&
    90→    f.optionB.trim().length > 0 &&
    91→    f.optionC.trim().length > 0 &&
    92→    f.optionD.trim().length > 0
    93→  )
    94→}

    96→export function ActivityEditorScreen() {
    97→  const navigate = useAppStore((s) => s.navigate)
    98→  const params = useAppStore((s) => s.params)
    99→  const activityId = params.activityId
   100→
   101→  const [activity, setActivity] = useState<ActivityDTO | null>(null)
   102→  const [loading, setLoading] = useState(true)
   103→  const [selectedId, setSelectedId] = useState<string | null>(null)
   104→  const [form, setForm] = useState<QuestionFormState | null>(null)
   105→  const [dirty, setDirty] = useState(false)
   106→  const [saving, setSaving] = useState(false)
   107→  const [publishOpen, setPublishOpen] = useState(false)
   108→  const [publishing, setPublishing] = useState(false)
   109→  const [publishResult, setPublishResult] = useState<ActivityDTO | null>(null)
   110→  const [deletingQuestion, setDeletingQuestion] = useState<QuestionDTO | null>(null)
   111→  const [titleDraft, setTitleDraft] = useState('')
   112→  const [savingTitle, setSavingTitle] = useState(false)
   113→  const titleInputRef = useRef<HTMLInputElement>(null)
   114→
   115→  const fetchActivity = useCallback(async () => {
   116→    if (!activityId) return
   117→    setLoading(true)
   118→    try {
   119→      const res = await api.get<{ activity: ActivityDTO }>(`/api/activities/${activityId}`)
   120→      setActivity(res.activity)
   121→      setTitleDraft(res.activity.title)
   122→      setSelectedId((prev) => {
   123→        const exists = res.activity.questions?.some((q) => q.id === prev)
   124→        if (exists) return prev
   125→        return res.activity.questions?.[0]?.id ?? null
   126→      })
   127→    } catch (err) {
   128→      const msg = err instanceof ApiError ? err.message : 'Failed to load activity'
   129→      toast.error(msg)
   130→      if (err instanceof ApiError && err.status === 401) {
   131→        navigate('admin-login')
   132→      } else if (err instanceof ApiError && err.status === 404) {
   133→        navigate('admin-dashboard')
   134→      }
   135→    } finally {
   136→      setLoading(false)
   137→    }
   138→  }, [activityId, navigate])
   139→
   140→  useEffect(() => {
   141→    void fetchActivity()
   142→  }, [fetchActivity])
   143→
   144→  // Sync form when selectedId changes.
   145→  useEffect(() => {
   146→    if (!activity || !selectedId) {
   147→      setForm(null)
   148→      setDirty(false)
   149→      return
   150→    }
   151→    const q = activity.questions?.find((x) => x.id === selectedId)
   152→    if (q) {
   153→      setForm(toFormState(q))
   154→      setDirty(false)
   155→    }
   156→  }, [selectedId, activity])
   157→
   158→  const selectedQuestion = useMemo(() => {
   159→    if (!activity || !selectedId) return null
   160→    return activity.questions?.find((q) => q.id === selectedId) ?? null
   161→  }, [activity, selectedId])
   162→
   163→  function updateField<K extends keyof QuestionFormState>(field: K, value: QuestionFormState[K]) {
   164→    if (!form) return
   165→    setForm({ ...form, [field]: value })
   166→    setDirty(true)
   167→  }
   168→
   169→  async function handleSaveQuestion() {
   170→    if (!form || !selectedQuestion) return
   171→    if (!isFormValid(form)) {
   172→      toast.error('Question text and all 4 options are required')
   173→      return
   174→    }
   175→    setSaving(true)
   176→    try {
   177→      const res = await api.patch<{ question: QuestionDTO }>(
   178→        `/api/questions/${selectedQuestion.id}`,
   179→        {
   180→          questionText: form.questionText,
   181→          optionA: form.optionA,
   182→          optionB: form.optionB,
   183→          optionC: form.optionC,
   184→          optionD: form.optionD,
   185→          correctOption: form.correctOption,
   186→          timeLimit: form.timeLimit,
   187→        },
   188→      )
   189→      setActivity((prev) =>
   190→        prev
   191→          ? {
   192→              ...prev,
   193→              questions: prev.questions?.map((q) =>
   194→                q.id === res.question.id ? res.question : q,
   195→              ),
   196→            }
   197→          : prev,
   198→      )
   199→      setDirty(false)
   200→      toast.success('Question saved')
   201→    } catch (err) {
   202→      const msg = err instanceof ApiError ? err.message : 'Failed to save question'
   203→      toast.error(msg)
   204→    } finally {
   205→      setSaving(false)
   206→    }
   207→  }
   208→
   209→  async function handleAddQuestion() {
   210→    if (!activity) return
   211→    if (activity.status !== 'DRAFT') {
   212→      toast.error('You can only add questions while the activity is a draft')
   213→      return
   214→    }
   215→    try {
   216→      const res = await api.post<{ question: QuestionDTO }>(
   217→        `/api/activities/${activity.id}/questions`,
   218→        {
   219→          questionText: 'New question',
   220→          optionA: 'Option A',
   221→          optionB: 'Option B',
   222→          optionC: 'Option C',
   223→          optionD: 'Option D',
   224→          correctOption: 'A',
   225→          timeLimit: 30,
   226→        },
   227→      )
   228→      setActivity((prev) =>
   229→        prev
   230→          ? { ...prev, questions: [...(prev.questions ?? []), res.question] }
   231→          : prev,
   232→      )
   233→      setSelectedId(res.question.id)
   234→      toast.success('Question added')
   235→    } catch (err) {
   236→      const msg = err instanceof ApiError ? err.message : 'Failed to add question'
   237→      toast.error(msg)
   238→    }
   239→  }
   240→
   241→  async function handleDeleteQuestion() {
   242→    if (!deletingQuestion || !activity) return
   243→    const targetId = deletingQuestion.id
   244→    try {
   245→      await api.delete(`/api/questions/${targetId}`)
   246→      const remaining = (activity.questions ?? []).filter((q) => q.id !== targetId)
   247→      setActivity({ ...activity, questions: remaining })
   248→      if (selectedId === targetId) {
   249→        setSelectedId(remaining[0]?.id ?? null)
   250→      }
   251→      toast.success('Question deleted')
   252→    } catch (err) {
   253→      const msg = err instanceof ApiError ? err.message : 'Failed to delete question'
   254→      toast.error(msg)
   255→    } finally {
   256→      setDeletingQuestion(null)
   257→    }
   258→  }
   259→
   260→  async function handleSaveTitle() {
   261→    if (!activity) return
   262→    if (titleDraft.trim() === activity.title) return
   263→    if (activity.status !== 'DRAFT') {
   264→      toast.error('Title can only be changed while the activity is a draft')
   265→      setTitleDraft(activity.title)
   266→      return
   267→    }
   268→    if (!titleDraft.trim()) {
   269→      setTitleDraft(activity.title)
   270→      return
   271→    }
   272→    setSavingTitle(true)
   273→    try {
   274→      const res = await api.patch<{ activity: ActivityDTO }>(`/api/activities/${activity.id}`, {
   275→        title: titleDraft.trim(),
   276→      })
   277→      setActivity((prev) => (prev ? { ...prev, title: res.activity.title } : prev))
   278→      toast.success('Title updated')
   279→    } catch (err) {
   280→      const msg = err instanceof ApiError ? err.message : 'Failed to update title'
   281→      toast.error(msg)
   282→      setTitleDraft(activity.title)
   283→    } finally {
   284→      setSavingTitle(false)
   285→    }
   286→  }
   287→
   288→  function handleTitleKeyDown(e: KeyboardEvent<HTMLInputElement>) {
   289→    if (e.key === 'Enter') {
   290→      e.preventDefault()
   291→      e.currentTarget.blur()
   292→    }
   293→  }
   294→
   295→  async function handlePublish() {
   296→    if (!activity) return
   297→    setPublishing(true)
   298→    try {
   299→      const res = await api.post<{ activity: ActivityDTO }>(
   300→        `/api/activities/${activity.id}/publish`,
   301→      )
   302→      setActivity(res.activity)
   303→      setPublishResult(res.activity)
   304→      toast.success('Activity published!')
   305→    } catch (err) {
   306→      const msg = err instanceof ApiError ? err.message : 'Failed to publish'
   307→      toast.error(msg)
   308→    } finally {
   309→      setPublishing(false)
   310→    }
   311→  }
   312→
   313→  function handleTitleSubmit(e: FormEvent) {
   314→    e.preventDefault()
   315→    titleInputRef.current?.blur()
   316→  }
   317→
   318→  async function copyAccessCode(code: string) {
   319→    try {
   320→      await navigator.clipboard.writeText(code)
   321→      toast.success('Access code copied')
   322→    } catch {
   323→      toast.error('Could not copy code')
   324→    }
   325→  }
   326→
   327→  // ---------- Loading ----------
   328→  if (loading) {
   329→    return (
   330→      <Shell>
   331→        <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
   332→          <Skeleton className="mb-6 h-12 w-72" />
   333→          <div className="grid gap-6 lg:grid-cols-[300px_1fr]">
   334→            <div className="space-y-3">
   335→              {Array.from({ length: 3 }).map((_, i) => (
   336→                <Skeleton key={i} className="h-16 w-full" />
   337→              ))}
   338→            </div>
   339→            <Skeleton className="h-96 w-full" />
   340→          </div>
   341→        </div>
   342→      </Shell>
   343→    )
   344→  }
   345→
   346→  if (!activity) {
   347→    return (
   348→      <Shell>
   349→        <div className="mx-auto max-w-7xl px-4 py-12 text-center sm:px-6 lg:px-8">
   350→          <p className="text-muted-foreground">Activity not found.</p>
   351→          <Button variant="ghost" onClick={() => navigate('admin-dashboard')} className="mt-4">
   352→            <ArrowLeft className="h-4 w-4" /> Back to dashboard
   353→          </Button>
   354→        </div>
   355→      </Shell>
   356→    )
   357→  }
   358→
   359→  const isDraft = activity.status === 'DRAFT'
   360→  const questionCount = activity.questions?.length ?? 0
   361→  const formInvalid = !isFormValid(form)
   362→
   363→  return (
   364→    <Shell>
   365→      {/* Top bar */}
   366→      <header className="sticky top-0 z-30 border-b border-border/60 bg-background/80 backdrop-blur">
   367→        <div className="mx-auto flex max-w-7xl items-center gap-3 px-4 py-3 sm:px-6 lg:px-8">
   368→          <Button
   369→            variant="ghost"
   370→            size="icon"
   371→            onClick={() => navigate('admin-dashboard')}
   372→            aria-label="Back to dashboard"
   373→          >
   374→            <ArrowLeft className="h-4 w-4" />
   375→          </Button>
   376→          <form className="relative min-w-0 flex-1" onSubmit={handleTitleSubmit}>
   377→            <Input
   378→              ref={titleInputRef}
   379→              value={titleDraft}
   380→              onChange={(e) => setTitleDraft(e.target.value)}
   381→              onBlur={handleSaveTitle}
   382→              onKeyDown={handleTitleKeyDown}
   383→              disabled={!isDraft || savingTitle}
   384→              className="h-10 border-transparent bg-transparent px-2 text-base font-semibold hover:border-input focus-visible:border-input sm:text-lg"
   385→              aria-label="Activity title"
   386→            />
   387→            {savingTitle && (
   388→              <Loader2 className="absolute right-2 top-1/2 h-3.5 w-3.5 -translate-y-1/2 animate-spin text-muted-foreground" />
   389→            )}
   390→          </form>
   391→          <Badge variant="outline" className={STATUS_BADGE_CLASS[activity.status]}>
   392→            {activity.status}
   393→          </Badge>
   394→          {/* Recommended max participants — DISPLAY ONLY. The join API
   395→              (`src/app/api/join/route.ts`) actually accepts up to 99, but the
   396→              admin-facing recommendation is 80 so hosts plan their room size
   397→              accordingly. See MAX_PARTICIPANTS_DISPLAY in
   398→              `src/lib/participant-icons.tsx`. */}
   399→          <Badge
   400→            variant="outline"
   401→            className="hidden items-center gap-1.5 border-primary/30 bg-primary/5 text-primary sm:inline-flex"
   402→            title={`Recommended room size — the join API accepts up to 99 participants`}
   403→          >
   404→            <Users className="h-3.5 w-3.5" />
   405→            Max {MAX_PARTICIPANTS_DISPLAY} participants
   406→          </Badge>
   407→          <div className="hidden items-center gap-2 sm:flex">
   408→            {activity.status === 'DRAFT' ? (
   409→              <Button
   410→                onClick={() => setPublishOpen(true)}
   411→                disabled={questionCount === 0 || publishing}
   412→                size="sm"
   413→              >
   414→                <Sparkles className="h-4 w-4" /> Publish
   415→              </Button>
   416→            ) : (
   417→              <div className="flex items-center gap-1 rounded-md border border-border bg-muted/40 px-2 py-1">
   418→                <span className="font-mono text-sm font-bold tracking-widest">
   419→                  {activity.accessCode ?? '------'}
   420→                </span>
   421→                <Button
   422→                  variant="ghost"
   423→                  size="icon"
   424→                  className="h-7 w-7"
   425→                  onClick={() => activity.accessCode && copyAccessCode(activity.accessCode)}
   426→                  aria-label="Copy access code"
   427→                >
   428→                  <Copy className="h-3.5 w-3.5" />
   429→                </Button>
   430→              </div>
   431→            )}
   432→            <Button
   433→              variant="secondary"
   434→              size="sm"
   435→              disabled={activity.status === 'DRAFT'}
   436→              onClick={() => navigate('admin-present', { activityId: activity.id })}
   437→            >
   438→              <Presentation className="h-4 w-4" /> Present
   439→            </Button>
   440→          </div>
   441→        </div>
   442→        {/* Mobile actions */}
   443→        <div className="flex items-center gap-2 border-t border-border/40 px-4 py-2 sm:hidden">
   444→          {activity.status === 'DRAFT' ? (
   445→            <Button
   446→              onClick={() => setPublishOpen(true)}
   447→              disabled={questionCount === 0 || publishing}
   448→              size="sm"
   449→              className="flex-1"
   450→            >
   451→              <Sparkles className="h-4 w-4" /> Publish
   452→            </Button>
   453→          ) : (
   454→            <Button
   455→              variant="outline"
   456→              size="sm"
   457→              className="flex-1"
   458→              onClick={() => activity.accessCode && copyAccessCode(activity.accessCode)}
   459→            >
   460→              <span className="font-mono font-bold tracking-widest">{activity.accessCode}</span>
   461→              <Copy className="h-3.5 w-3.5" />
   462→            </Button>
   463→          )}
   464→          <Button
   465→            variant="secondary"
   466→            size="sm"
   467→            disabled={activity.status === 'DRAFT'}
   468→            onClick={() => navigate('admin-present', { activityId: activity.id })}
   469→          >
   470→            <Presentation className="h-4 w-4" />
   471→          </Button>
   472→        </div>
   473→      </header>
   474→
   475→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-6 sm:px-6 lg:px-8">
   476→        <div className="grid gap-6 lg:grid-cols-[300px_1fr]">
   477→          {/* Question list */}
   478→          <aside className="lg:sticky lg:top-28 lg:self-start">
   479→            <div className="mb-2 flex items-center justify-between">
   480→              <h2 className="text-sm font-semibold text-muted-foreground">
   481→                Questions ({questionCount})
   482→              </h2>
   483→            </div>
   484→            <ScrollArea className="h-[calc(100vh-14rem)] pr-3">
   485→              <div className="space-y-2">
   486→                {(activity.questions ?? []).map((q, i) => {
   487→                  const isSelected = q.id === selectedId
   488→                  return (
   489→                    <button
   490→                      key={q.id}
   491→                      onClick={() => setSelectedId(q.id)}
   492→                      className={`group relative flex w-full items-start gap-3 rounded-xl border-2 px-3 py-2.5 text-left transition-all ${
   493→                        isSelected
   494→                          ? 'border-primary bg-primary/5'
   495→                          : 'border-border bg-card hover:border-primary/40 hover:bg-accent/40'
   496→                      }`}
   497→                    >
   498→                      <span
   499→                        className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-xs font-bold ${
   500→                          isSelected
   501→                            ? 'bg-primary text-primary-foreground'
   502→                            : 'bg-muted text-muted-foreground'
   503→                        }`}
   504→                      >
   505→                        {i + 1}
   506→                      </span>
   507→                      <div className="min-w-0 flex-1">
   508→                        <p className="truncate text-sm font-medium">
   509→                          {q.questionText || (
   510→                            <span className="italic text-muted-foreground">Untitled question</span>
   511→                          )}
   512→                        </p>
   513→                        <p className="mt-0.5 text-xs text-muted-foreground">
   514→                          {q.timeLimit}s · correct: {q.correctOption}
   515→                        </p>
   516→                      </div>
   517→                      {isDraft && (
   518→                        <span
   519→                          role="button"
   520→                          tabIndex={0}
   521→                          aria-label={`Delete question ${i + 1}`}
   522→                          onClick={(e) => {
   523→                            e.stopPropagation()
   524→                            setDeletingQuestion(q)
   525→                          }}
   526→                          onKeyDown={(e) => {
   527→                            if (e.key === 'Enter' || e.key === ' ') {
   528→                              e.preventDefault()
   529→                              e.stopPropagation()
   530→                              setDeletingQuestion(q)
   531→                            }
   532→                          }}
   533→                          className="absolute right-1.5 top-1.5 hidden h-7 w-7 items-center justify-center rounded-md text-muted-foreground hover:bg-destructive/10 hover:text-destructive group-hover:flex"
   534→                        >
   535→                          <Trash2 className="h-3.5 w-3.5" />
   536→                        </span>
   537→                      )}
   538→                    </button>
   539→                  )
   540→                })}
   541→                {questionCount === 0 && (
   542→                  <p className="px-3 py-6 text-center text-xs text-muted-foreground">
   543→                    No questions yet. Add your first one!
   544→                  </p>
   545→                )}
   546→              </div>
   547→              {isDraft && (
   548→                <Button
   549→                  onClick={handleAddQuestion}
   550→                  variant="outline"
   551→                  className="mt-3 w-full border-dashed"
   552→                >
   553→                  <Plus className="h-4 w-4" /> Add question
   554→                </Button>
   555→              )}
   556→            </ScrollArea>
   557→          </aside>
   558→
   559→          {/* Editor */}
   560→          <section>
   561→            <AnimatePresence mode="wait">
   562→              {selectedQuestion && form ? (
   563→                <motion.div
   564→                  key={selectedQuestion.id}
   565→                  initial={{ opacity: 0, y: 10 }}
   566→                  animate={{ opacity: 1, y: 0 }}
   567→                  exit={{ opacity: 0, y: -10 }}
   568→                  transition={{ duration: 0.2 }}
   569→                >
   570→                  <Card>
   571→                    <CardHeader className="flex flex-row items-start justify-between gap-2 space-y-0">
   572→                      <div>
   573→                        <CardTitle className="flex items-center gap-2">
   574→                          <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-primary/15 text-sm font-bold text-primary">
   575→                            {selectedQuestion.questionOrder}
   576→                          </span>
   577→                          Edit question
   578→                        </CardTitle>
   579→                        <CardDescription>
   580→                          Keep it clear and concise. Tap a letter badge to mark the correct answer.
   581→                        </CardDescription>
   582→                      </div>
   583→                      {dirty && (
   584→                        <Badge
   585→                          variant="outline"
   586→                          className="border-amber-500/40 text-amber-600 dark:text-amber-400"
   587→                        >
   588→                          Unsaved changes
   589→                        </Badge>
   590→                      )}
   591→                    </CardHeader>
   592→                    <CardContent className="space-y-6">
   593→                      <div className="space-y-2">
   594→                        <Label htmlFor="questionText">Question</Label>
   595→                        <Textarea
   596→                          id="questionText"
   597→                          value={form.questionText}
   598→                          onChange={(e) => updateField('questionText', e.target.value)}
   599→                          placeholder="Type your question…"
   600→                          rows={3}
   601→                          disabled={!isDraft}
   602→                        />
   603→                      </div>
   604→
   605→                      <div className="space-y-3">
   606→                        <Label>Options</Label>
   607→                        <div className="space-y-2">
   608→                          {OPTION_KEYS.map((key) => {
   609→                            const field = `option${key}` as
   610→                              | 'optionA'
   611→                              | 'optionB'
   612→                              | 'optionC'
   613→                              | 'optionD'
   614→                            const isCorrect = form.correctOption === key
   615→                            return (
   616→                              <div
   617→                                key={key}
   618→                                className={`flex items-center gap-2 rounded-xl border-2 px-2 py-1.5 transition-colors ${
   619→                                  isCorrect
   620→                                    ? 'border-emerald-500/60 bg-emerald-500/5'
   621→                                    : 'border-border bg-card'
   622→                                }`}
   623→                              >
   624→                                <button
   625→                                  type="button"
   626→                                  onClick={() => updateField('correctOption', key)}
   627→                                  disabled={!isDraft}
   628→                                  aria-label={`Mark option ${key} as correct`}
   629→                                  aria-pressed={isCorrect}
   630→                                  className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-lg text-sm font-bold transition-colors ${
   631→                                    isCorrect
   632→                                      ? 'bg-emerald-500 text-white'
   633→                                      : 'bg-muted text-muted-foreground hover:bg-primary/15 hover:text-primary'
   634→                                  }`}
   635→                                >
   636→                                  {isCorrect ? <Check className="h-4 w-4" /> : key}
   637→                                </button>
   638→                                <Input
   639→                                  value={form[field]}
   640→                                  onChange={(e) => updateField(field, e.target.value)}
   641→                                  placeholder={`Option ${key}`}
   642→                                  disabled={!isDraft}
   643→                                  className="h-10 border-0 bg-transparent shadow-none focus-visible:ring-0"
   644→                                />
   645→                              </div>
   646→                            )
   647→                          })}
   648→                        </div>
   649→                        <p className="text-xs text-muted-foreground">
   650→                          Correct answer:{' '}
   651→                          <span className="font-semibold text-emerald-600 dark:text-emerald-400">
   652→                            {form.correctOption}
   653→                          </span>
   654→                        </p>
   655→                      </div>
   656→
   657→                      <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
   658→                        <div className="space-y-2">
   659→                          <Label htmlFor="timeLimit">Time limit</Label>
   660→                          <Select
   661→                            value={String(form.timeLimit)}
   662→                            onValueChange={(v) => updateField('timeLimit', Number(v))}
   663→                            disabled={!isDraft}
   664→                          >
   665→                            <SelectTrigger id="timeLimit" className="w-40">
   666→                              <SelectValue />
   667→                            </SelectTrigger>
   668→                            <SelectContent>
   669→                              {TIME_LIMITS.map((t) => (
   670→                                <SelectItem key={t} value={String(t)}>
   671→                                  {t} seconds
   672→                                </SelectItem>
   673→                              ))}
   674→                            </SelectContent>
   675→                          </Select>
   676→                        </div>
   677→
   678→                        <div className="flex items-center gap-2">
   679→                          {isDraft && (
   680→                            <Button
   681→                              variant="ghost"
   682→                              onClick={() => setDeletingQuestion(selectedQuestion)}
   683→                              className="text-destructive hover:bg-destructive/10 hover:text-destructive"
   684→                            >
   685→                              <Trash2 className="h-4 w-4" /> Delete
   686→                            </Button>
   687→                          )}
   688→                          <Button
   689→                            onClick={handleSaveQuestion}
   690→                            disabled={!isDraft || formInvalid || saving || !dirty}
   691→                          >
   692→                            {saving ? (
   693→                              <Loader2 className="h-4 w-4 animate-spin" />
   694→                            ) : (
   695→                              <Save className="h-4 w-4" />
   696→                            )}
   697→                            Save
   698→                          </Button>
   699→                        </div>
   700→                      </div>
   701→                    </CardContent>
   702→                  </Card>
   703→                </motion.div>
   704→              ) : (
   705→                <motion.div
   706→                  key="empty"
   707→                  initial={{ opacity: 0 }}
   708→                  animate={{ opacity: 1 }}
   709→                  exit={{ opacity: 0 }}
   710→                  className="flex h-64 flex-col items-center justify-center rounded-2xl border-2 border-dashed border-border/60 bg-card/40 text-center"
   711→                >
   712→                  <Sparkles className="h-10 w-10 text-muted-foreground/60" />
   713→                  <p className="mt-3 text-sm text-muted-foreground">
   714→                    Select a question on the left to edit, or add a new one.
   715→                  </p>
   716→                  {isDraft && (
   717→                    <Button onClick={handleAddQuestion} variant="outline" className="mt-4">
   718→                      <Plus className="h-4 w-4" /> Add first question
   719→                    </Button>
   720→                  )}
   721→                </motion.div>
   722→              )}
   723→            </AnimatePresence>
   724→          </section>
   725→        </div>
   726→      </main>
   727→
   728→      {/* Publish dialog */}
   729→      <Dialog
   730→        open={publishOpen}
   731→        onOpenChange={(o) => {
   732→          setPublishOpen(o)
   733→          if (!o) setPublishResult(null)
   734→        }}
   735→      >
   736→        <DialogContent>
   737→          {publishResult ? (
   738→            <>
   739→              <DialogHeader>
   740→                <DialogTitle>Activity published!</DialogTitle>
   741→                <DialogDescription>
   742→                  Share this access code with your participants. They can join from the participant entrance.
   743→                </DialogDescription>
   744→              </DialogHeader>
   745→              <div className="rounded-2xl border-2 border-dashed border-primary/30 bg-primary/5 p-6 text-center">
   746→                <p className="text-xs font-semibold uppercase tracking-widest text-muted-foreground">
   747→                  Access code
   748→                </p>
   749→                <p className="mt-2 font-mono text-5xl font-bold tracking-[0.3em] text-primary">
   750→                  {publishResult.accessCode}
   751→                </p>
   752→                <Button
   753→                  variant="outline"
   754→                  size="sm"
   755→                  className="mt-3"
   756→                  onClick={() =>
   757→                    publishResult.accessCode && copyAccessCode(publishResult.accessCode)
   758→                  }
   759→                >
   760→                  <Copy className="h-3.5 w-3.5" /> Copy code
   761→                </Button>
   762→              </div>
   763→              <DialogFooter className="gap-2">
   764→                <Button variant="ghost" onClick={() => setPublishOpen(false)}>
   765→                  Stay in editor
   766→                </Button>
   767→                <Button
   768→                  onClick={() => navigate('admin-present', { activityId: publishResult.id })}
   769→                >
   770→                  <Presentation className="h-4 w-4" /> Present now
   771→                </Button>
   772→              </DialogFooter>
   773→            </>
   774→          ) : (
   775→            <>
   776→              <DialogHeader>
   777→                <DialogTitle>Publish activity?</DialogTitle>
   778→                <DialogDescription>
   779→                  This will lock the activity and generate a 6-digit access code. You won&apos;t be
   780→                  able to edit questions after publishing.
   781→                </DialogDescription>
   782→              </DialogHeader>
   783→              <div className="rounded-xl bg-muted/40 p-3 text-sm">
   784→                <div className="flex items-center justify-between">
   785→                  <span className="text-muted-foreground">Title</span>
   786→                  <span className="truncate pl-2 text-right font-medium">{activity.title}</span>
   787→                </div>
   788→                <Separator className="my-2" />
   789→                <div className="flex items-center justify-between">
   790→                  <span className="text-muted-foreground">Questions</span>
   791→                  <span className="font-medium">{questionCount}</span>
   792→                </div>
   793→                <Separator className="my-2" />
   794→                <div className="flex items-center justify-between">
   795→                  <span className="text-muted-foreground">Max participants</span>
   796→                  <span className="font-medium">{MAX_PARTICIPANTS_DISPLAY}</span>
   797→                </div>
   798→              </div>
   799→              <DialogFooter className="gap-2">
   800→                <Button variant="ghost" onClick={() => setPublishOpen(false)} disabled={publishing}>
   801→                  Cancel
   802→                </Button>
   803→                <Button onClick={handlePublish} disabled={publishing}>
   804→                  {publishing ? (
   805→                    <Loader2 className="h-4 w-4 animate-spin" />
   806→                  ) : (
   807→                    <Sparkles className="h-4 w-4" />
   808→                  )}
   809→                  Publish now
   810→                </Button>
   811→              </DialogFooter>
   812→            </>
   813→          )}
   814→        </DialogContent>
   815→      </Dialog>
   816→
   817→      {/* Delete question confirm */}
   818→      <AlertDialog
   819→        open={!!deletingQuestion}
   820→        onOpenChange={(o) => !o && setDeletingQuestion(null)}
   821→      >
   822→        <AlertDialogContent>
   823→          <AlertDialogHeader>
   824→            <AlertDialogTitle>
   825→              Delete question {deletingQuestion?.questionOrder}?
   826→            </AlertDialogTitle>
   827→            <AlertDialogDescription>
   828→              This cannot be undone. Remaining questions will be renumbered automatically.
   829→            </AlertDialogDescription>
   830→          </AlertDialogHeader>
   831→          <AlertDialogFooter>
   832→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   833→            <AlertDialogAction
   834→              onClick={handleDeleteQuestion}
   835→              className="bg-destructive text-white hover:bg-destructive/90"
   836→            >
   837→              Delete
   838→            </AlertDialogAction>
   839→          </AlertDialogFooter>
   840→        </AlertDialogContent>
   841→      </AlertDialog>
   842→    </Shell>
   843→  )
   844→}
   845→
   846→function Shell({ children }: { children: React.ReactNode }) {
   847→  return (
   848→    <div className="flex min-h-screen flex-col bg-background">
   849→      {children}
   850→      <AppFooter />
   851→    </div>
   852→  )
   853→}
