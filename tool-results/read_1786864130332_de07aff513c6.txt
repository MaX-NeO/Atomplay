    64→// 70-color palette generated by hue rotation around the full color wheel.
    65→// Each entry has a vivid border + matching soft glow + readable text color.
    66→const PARTICIPANT_COLORS: { border: string; glow: string; text: string; soft: string }[] =
    67→  Array.from({ length: 70 }, (_, i) => {
    68→    const hue = Math.round((i * 360) / 70)
    69→    return {
    70→      border: `hsl(${hue} 85% 65%)`,
    71→      glow: `hsl(${hue} 85% 55% / 0.45)`,
    72→      text: `hsl(${hue} 90% 82%)`,
    73→      soft: `hsl(${hue} 85% 55% / 0.16)`,
    74→    }
    75→  })

    77→// Deterministic color assignment — mix the name hash with the index so visually
    78→// nearby bubbles don't always clash, and the same person always gets the same
    79→// color across reloads.
    80→function colorForParticipant(name: string, idx: number) {
    81→  let hash = 0
    82→  for (let i = 0; i < name.length; i++) hash = (hash * 31 + name.charCodeAt(i)) | 0
    83→  return PARTICIPANT_COLORS[Math.abs(hash + idx * 7) % PARTICIPANT_COLORS.length]
    84→}

    86→// Phyllotaxis "sunflower" layout — the first participant is pinned to the
    87→// center as a large anchor bubble, and each subsequent joiner spawns around
    88→// them at the golden angle (137.5°). The cluster stays balanced as it grows
    89→// and every outside bubble gets its own size + independent drift animation
    90→// so the whole field shimmers gently instead of rigidly rotating.
    91→const GOLDEN_ANGLE = Math.PI * (3 - Math.sqrt(5)) // ~2.39996 rad = 137.5°
    92→
    93→const CENTER_BUBBLE_DIAMETER = 132 // first user — large anchor
    94→const OUTSIDE_BUBBLE_MIN = 44 // smallest outside circle
    95→const OUTSIDE_BUBBLE_MAX = 72 // largest outside circle
    96→const DRIFT_AMPLITUDE = 6 // ±px movement for outside bubbles
    97→
    98→function phyllotaxisPosition(i: number): { x: number; y: number } {
    99→  if (i === 0) return { x: 0, y: 0 } // first user — dead center
   100→  const r = 78 + 24 * Math.sqrt(i)
   101→  const theta = i * GOLDEN_ANGLE
   102→  return { x: Math.cos(theta) * r, y: Math.sin(theta) * r }
   103→}
   104→
   105→// Deterministic pseudo-random in [0, 1) from a string seed — keeps each
   106→// bubble's size + drift stable across reloads and re-renders.
   107→function seededRand(seed: string): number {
   108→  let h = 0
   109→  for (let i = 0; i < seed.length; i++) h = (h * 33 + seed.charCodeAt(i)) | 0
   110→  return (Math.abs(h) % 10000) / 10000
   111→}
   112→
   113→function bubbleDiameter(name: string, idx: number): number {
   114→  const r = seededRand(`${name}-${idx}-size`)
   115→  return OUTSIDE_BUBBLE_MIN + r * (OUTSIDE_BUBBLE_MAX - OUTSIDE_BUBBLE_MIN)
   116→}
   117→
   118→// Each outside bubble drifts on its own tiny loop so the cluster feels alive
   119→// without any big rotation. Returns x/y keyframes + loop timing.
   120→function bubbleDrift(name: string, idx: number) {
   121→  const r1 = seededRand(`${name}-${idx}-x1`)
   122→  const r2 = seededRand(`${name}-${idx}-x2`)
   123→  const r3 = seededRand(`${name}-${idx}-y1`)
   124→  const r4 = seededRand(`${name}-${idx}-y2`)
   125→  const r5 = seededRand(`${name}-${idx}-dur`)
   126→  const r6 = seededRand(`${name}-${idx}-delay`)
   127→  return {
   128→    x: [0, (r1 - 0.5) * 2 * DRIFT_AMPLITUDE, 0, (r2 - 0.5) * 2 * DRIFT_AMPLITUDE, 0],
   129→    y: [0, (r3 - 0.5) * 2 * DRIFT_AMPLITUDE, 0, (r4 - 0.5) * 2 * DRIFT_AMPLITUDE, 0],
   130→    duration: 5 + r5 * 4, // 5..9s
   131→    delay: r6 * 2, // 0..2s
   132→  }
   133→}

   135→// Adaptive scale — keeps the phyllotaxis cluster fitting inside the viewport as
   136→// the participant count grows. Computed from the outermost bubble's distance
   137→// + max bubble size, clamped to [0.4, 1] so it never gets unreadable or huge.
   138→function computeClusterScale(count: number, vw: number, vh: number): number {
   139→  const outer = 78 + 24 * Math.sqrt(Math.max(0, count - 1)) + OUTSIDE_BUBBLE_MAX / 2
   140→  const diameter = outer * 2
   141→  // Leave room for header (~180px) + start button (~90px) + side padding.
   142→  const avail = Math.min(vw - 64, vh - 280)
   143→  if (avail <= 0) return 0.5
   144→  const raw = avail / diameter
   145→  return Math.max(0.4, Math.min(1, raw))
   146→}

   156→  return {
   157→    questionId: q.id,
   158→    questionOrder: q.questionOrder,

   161→    options: [
   162→      { key: 'A', label: q.optionA },

   164→      { key: 'C', label: q.optionC },

   171→function labelsFromLive(q: LiveQuestion | null): Partial<Record<OptionKey, string>> {
   172→  if (!q) return {}
   173→  const out: Partial<Record<OptionKey, string>> = {}
   174→  for (const o of q.options) out[o.key] = o.label
   175→  return out
