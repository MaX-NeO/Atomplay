    69→    69→// 70-color palette generated by hue rotation around the full color wheel.
    70→    70→// Each entry has a vivid border + matching soft glow + readable text color.
    71→    71→const PARTICIPANT_COLORS: { border: string; glow: string; text: string; soft: string }[] =
    72→    72→  Array.from({ length: 70 }, (_, i) => {
    73→    73→    const hue = Math.round((i * 360) / 70)
    74→    74→    return {
    75→    75→      border: `hsl(${hue} 85% 65%)`,
    76→    76→      glow: `hsl(${hue} 85% 55% / 0.45)`,
    77→    77→      text: `hsl(${hue} 90% 82%)`,
    78→    78→      soft: `hsl(${hue} 85% 55% / 0.16)`,
    79→    79→    }
    80→    80→  })
    81→    81→
    82→    82→// Deterministic color assignment — mix the name hash with the index so visually
    83→    83→// nearby bubbles don't always clash, and the same person always gets the same
    84→    84→// color across reloads.
    85→    85→function colorForParticipant(name: string, idx: number) {
    86→    86→  let hash = 0
    87→    87→  for (let i = 0; i < name.length; i++) hash = (hash * 31 + name.charCodeAt(i)) | 0
    88→    88→  return PARTICIPANT_COLORS[Math.abs(hash + idx * 7) % PARTICIPANT_COLORS.length]
    89→    89→}

    91→    91→// Phyllotaxis "sunflower" layout — the first participant is pinned to the
    92→    92→// center as a large anchor bubble, and each subsequent joiner spawns around
    93→    93→// them at the golden angle (137.5°). The cluster stays balanced as it grows
    94→    94→// and every outside bubble gets its own size + independent drift animation
    95→    95→// so the whole field shimmers gently instead of rigidly rotating.
    96→    96→const GOLDEN_ANGLE = Math.PI * (3 - Math.sqrt(5)) // ~2.39996 rad = 137.5°
    97→    97→
    98→    98→const CENTER_BUBBLE_DIAMETER = 132 // first user — large anchor
    99→    99→const OUTSIDE_BUBBLE_MIN = 44 // smallest outside circle
   100→   100→const OUTSIDE_BUBBLE_MAX = 72 // largest outside circle
   101→   101→const DRIFT_AMPLITUDE = 6 // ±px movement for outside bubbles
   102→   102→
   103→   103→function phyllotaxisPosition(i: number): { x: number; y: number } {
   104→   104→  if (i === 0) return { x: 0, y: 0 } // first user — dead center
   105→   105→  const r = 78 + 24 * Math.sqrt(i)
   106→   106→  const theta = i * GOLDEN_ANGLE
   107→   107→  return { x: Math.cos(theta) * r, y: Math.sin(theta) * r }
   108→   108→}
   109→   109→
   110→   110→// Deterministic pseudo-random in [0, 1) from a string seed — keeps each
   111→   111→// bubble's size + drift stable across reloads and re-renders.
   112→   112→function seededRand(seed: string): number {
   113→   113→  let h = 0
   114→   114→  for (let i = 0; i < seed.length; i++) h = (h * 33 + seed.charCodeAt(i)) | 0
   115→   115→  return (Math.abs(h) % 10000) / 10000
   116→   116→}
   117→   117→
   118→   118→function bubbleDiameter(name: string, idx: number): number {
   119→   119→  const r = seededRand(`${name}-${idx}-size`)
   120→   120→  return OUTSIDE_BUBBLE_MIN + r * (OUTSIDE_BUBBLE_MAX - OUTSIDE_BUBBLE_MIN)
   121→   121→}
   122→   122→
   123→   123→// Each outside bubble drifts on its own tiny loop so the cluster feels alive
   124→   124→// without any big rotation. Returns x/y keyframes + loop timing.
   125→   125→function bubbleDrift(name: string, idx: number) {
   126→   126→  const r1 = seededRand(`${name}-${idx}-x1`)
   127→   127→  const r2 = seededRand(`${name}-${idx}-x2`)
   128→   128→  const r3 = seededRand(`${name}-${idx}-y1`)
   129→   129→  const r4 = seededRand(`${name}-${idx}-y2`)
   130→   130→  const r5 = seededRand(`${name}-${idx}-dur`)
   131→   131→  const r6 = seededRand(`${name}-${idx}-delay`)
   132→   132→  return {
   133→   133→    x: [0, (r1 - 0.5) * 2 * DRIFT_AMPLITUDE, 0, (r2 - 0.5) * 2 * DRIFT_AMPLITUDE, 0],
   134→   134→    y: [0, (r3 - 0.5) * 2 * DRIFT_AMPLITUDE, 0, (r4 - 0.5) * 2 * DRIFT_AMPLITUDE, 0],
   135→   135→    duration: 5 + r5 * 4, // 5..9s
   136→   136→    delay: r6 * 2, // 0..2s
   137→   137→  }
   138→   138→}
   139→   139→
   140→   140→// Adaptive scale — keeps the phyllotaxis cluster fitting inside the viewport as
   141→   141→// the participant count grows. Computed from the outermost bubble's distance
   142→   142→// + max bubble size, clamped to [0.4, 1] so it never gets unreadable or huge.
   143→   143→function computeClusterScale(count: number, vw: number, vh: number): number {
   144→   144→  const outer = 78 + 24 * Math.sqrt(Math.max(0, count - 1)) + OUTSIDE_BUBBLE_MAX / 2
   145→   145→  const diameter = outer * 2
   146→   146→  // Leave room for header (~180px) + start button (~90px) + side padding.
   147→   147→  const avail = Math.min(vw - 64, vh - 280)
   148→   148→  if (avail <= 0) return 0.5
   149→   149→  const raw = avail / diameter
   150→   150→  return Math.max(0.4, Math.min(1, raw))
   151→   151→}
