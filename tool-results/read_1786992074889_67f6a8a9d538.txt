    99→function StatusBadge({ status }: { status: ActivityStatus }) {
   100→  if (status === 'LIVE') {
   101→    return (
   102→      <Badge className="gap-1.5 rounded-full border-amber-500/30 bg-amber-500/15 text-amber-600 dark:text-amber-400">
   103→        <span className="relative flex h-2 w-2">
   104→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-amber-500/80" />
   105→          <span className="relative inline-flex h-2 w-2 rounded-full bg-amber-500" />
   106→        </span>
   107→        LIVE
   108→      </Badge>
   109→    )
   110→  }
   111→  if (status === 'PUBLISHED') {
   112→    return (
   113→      <Badge className="gap-1.5 rounded-full border-emerald-500/30 bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   114→        <CircleDot className="h-3 w-3" />
   115→        Published
   116→      </Badge>
   117→    )
   118→  }
   119→  if (status === 'COMPLETED') {
   120→    return (
   121→      <Badge
   122→        variant="secondary"
   123→        className="gap-1.5 rounded-full bg-muted text-muted-foreground"
   124→      >
   125→        Completed
   126→      </Badge>
   127→    )
   128→  }
   129→  return (
   130→    <Badge
   131→      variant="secondary"
   132→      className="gap-1.5 rounded-full bg-muted text-muted-foreground"
   133→    >
   134→      <FileText className="h-3 w-3" />
   135→      Draft
   136→    </Badge>
   137→  )
   138→}

   140→function initials(name: string) {
   141→  return name
   142→    .split(' ')
   143→    .map((n) => n[0])
   144→    .slice(0, 2)
   145→    .join('')
   146→    .toUpperCase()
   147→}

   149→function greeting() {
   150→  const h = new Date().getHours()
   151→  if (h < 12) return 'morning'
   152→  if (h < 18) return 'afternoon'
   153→  return 'evening'
   154→}

   172→
   173→  // Delete confirm

   175→  const [deleting, setDeleting] = useState(false)
   176→
   177→  // View access code
   178→  const [codeActivity, setCodeActivity] = useState<ActivityWithCount | null>(null)
   179→

   181→  const [cloneTarget, setCloneTarget] = useState<ActivityWithCount | null>(null)
   182→  const [cloneTitle, setCloneTitle] = useState('')
   183→  const [clonePublishNow, setClonePublishNow] = useState(true)
   184→  const [cloning, setCloning] = useState(false)
   185→
   186→  // Reset dialog (only for COMPLETED)
   187→  const [resetTarget, setResetTarget] = useState<ActivityWithCount | null>(null)

   189→  const [resetting, setResetting] = useState(false)

   206→  async function fetchActivities() {
   207→    setLoading(true)
   208→    try {
   209→      const res = await api.get<{ activities: ActivityWithCount[] }>(
   210→        '/api/activities',
   211→      )
   212→      setActivities(res.activities)
   213→    } catch {
   214→      toast.error('Failed to load activities')
   215→    } finally {
   216→      setLoading(false)
   217→    }
   218→  }
   219→
   220→  const filtered = useMemo(() => {
   221→    return activities.filter((a) => {
   222→      if (filter !== 'all' && a.status !== filter) return false
   223→      if (search && !a.title.toLowerCase().includes(search.toLowerCase()))
   224→        return false
   225→      return true
   226→    })
   227→  }, [activities, filter, search])
   228→
   229→  const stats = useMemo(() => {
   230→    return {
   231→      total: activities.length,
   232→      liveOrPublished: activities.filter(
   233→        (a) => a.status === 'PUBLISHED' || a.status === 'LIVE',
   234→      ).length,
   235→      participants: activities.reduce(
   236→        (sum, a) =>
   237→          sum + (a._count?.participants ?? a.participantCount ?? 0),
   238→        0,
   239→      ),
   240→    }
   241→  }, [activities])
   242→
   243→  async function handleCreate() {
   244→    if (!newTitle.trim()) return
   245→    setCreating(true)
   246→    try {
   247→      const res = await api.post<{ activity: ActivityDTO }>('/api/activities', {
   248→        title: newTitle.trim(),
   249→        description: newDesc.trim(),
   250→      })
   251→      toast.success('Activity created')
   252→      setCreateOpen(false)
   253→      setNewTitle('')
   254→      setNewDesc('')
   255→      navigate('admin-editor', { activityId: res.activity.id })
   256→    } catch (err) {
   257→      const msg = err instanceof ApiError ? err.message : 'Failed to create activity'
   258→      toast.error(msg)
   259→    } finally {
   260→      setCreating(false)
   261→    }
   262→  }
   263→
   264→  async function handleDelete() {
   265→    if (!deleteTarget) return
   266→    setDeleting(true)
   267→    try {
   268→      await api.delete(`/api/activities/${deleteTarget.id}`)
   269→      toast.success(`“${deleteTarget.title}” deleted`)
   270→      setActivities((prev) => prev.filter((a) => a.id !== deleteTarget.id))
   271→    } catch (err) {
   272→      const msg = err instanceof ApiError ? err.message : 'Failed to delete activity'
   273→      toast.error(msg)
   274→    } finally {
   275→      setDeleting(false)
   276→      setDeleteTarget(null)
   277→    }
   278→  }

   280→  function openClone(activity: ActivityWithCount) {
   281→    setCloneTarget(activity)
   282→    setCloneTitle(
   283→      activity.title.toLowerCase().endsWith('(copy)')
   284→        ? activity.title
   285→        : `${activity.title} (Copy)`,
   286→    )
   287→    setClonePublishNow(true)
   288→  }
   289→
   290→  async function handleClone() {
   291→    if (!cloneTarget) return
   292→    setCloning(true)
   293→    try {
   294→      const res = await api.post<{
   295→        activity: ActivityDTO
   296→        warning: string | null
   297→      }>(`/api/activities/${cloneTarget.id}/clone`, {
   298→        title: cloneTitle.trim(),
   299→        publishImmediately: clonePublishNow,
   300→      })
   301→      if (res.warning) {
   302→        toast.warning(res.warning)
   303→      } else if (clonePublishNow && res.activity.accessCode) {
   304→        toast.success(
   305→          `Cloned & published — new code ${res.activity.accessCode}`,
   306→        )
   307→      } else {
   308→        toast.success(`Cloned as draft — “${res.activity.title}”`)
   309→      }
   310→      setCloneTarget(null)
   311→      // Refresh list so the new clone appears in the correct tab.
   312→      void fetchActivities()
   313→    } catch (err) {
   314→      const msg = err instanceof ApiError ? err.message : 'Failed to clone activity'
   315→      toast.error(msg)
   316→    } finally {
   317→      setCloning(false)
   318→    }
   319→  }
   320→
   321→  async function handleReset() {
   322→    if (!resetTarget) return
   323→    setResetting(true)
   324→    try {
   325→      const res = await api.post<{ activity: ActivityDTO }>(
   326→        `/api/activities/${resetTarget.id}/reset`,
   327→        { regenerateAccessCode: resetMode === 'regenerate' },
   328→      )
   329→      toast.success(
   330→        `Reset complete — access code ${res.activity.accessCode ?? '—'}`,
   331→      )
   332→      setResetTarget(null)
   333→      void fetchActivities()
   334→    } catch (err) {
   335→      const msg = err instanceof ApiError ? err.message : 'Failed to reset activity'
   336→      toast.error(msg)
   337→    } finally {
   338→      setResetting(false)
   339→    }
   340→  }

   576→      {/* Create dialog */}
